function closedDiagnostic(diagnostic, subcheck) {
  assert.ok(diagnostic);
  assert.equal(diagnostic.subcheck, subcheck);
  assert.deepEqual(Object.keys(diagnostic).sort(), ['cause', 'repair', 'schema', 'subcheck']);
  assert.deepEqual(Object.keys(diagnostic.repair).sort(), ['authority', 'automatic', 'kind']);
  assert.equal(diagnostic.repair.automatic, false);
  assert.equal(diagnostic.repair.authority, 'diagnostic-only');
  assert.equal(Buffer.byteLength(JSON.stringify(diagnostic)) <= 512, true);
  assert.equal(Object.isFrozen(diagnostic), true);
}
function mutate(path, change) {
  const original = readFileSync(path);
  const value = JSON.parse(original);
  change(value);
  writeFileSync(path, canonicalJson(value));
  return () => writeFileSync(path, original);
}
