import os from "node:os";
import { mkdirSync } from "node:fs";
import { syncBuiltinESMExports } from "node:module";
import { dirname, resolve } from "node:path";
import { fileURLToPath } from "node:url";

const here = dirname(fileURLToPath(import.meta.url));
const home = resolve(here, "synthetic-home");
const temp = resolve(here, "test-tmp");
mkdirSync(home, { recursive: true, mode: 0o700 });
mkdirSync(temp, { recursive: true, mode: 0o700 });
process.env.HOME = home;
process.env.USERPROFILE = home;
process.env.TMPDIR = temp;
os.userInfo = () => ({ username: "g33-synthetic", uid: 61931, gid: 61931, shell: "/bin/sh", homedir: home });
syncBuiltinESMExports();
process.env.NODE_OPTIONS = `--import=${fileURLToPath(import.meta.url)}`;
