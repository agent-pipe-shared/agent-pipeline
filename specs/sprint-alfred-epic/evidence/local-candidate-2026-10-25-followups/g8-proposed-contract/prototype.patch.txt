+// SPDX-License-Identifier: SUL-1.0
+/** Inert G8 contract prototype. Preparation is a proposal, never review evidence.
+ * No execution port, identity issuer, receipt writer or admission authority is exported.
+ */
+import { createHash } from "node:crypto";
+import { canonical } from "./po-approval-proof.mjs";
+import { inspectSpecCriterionRows } from "./critic-accepted-contract.mjs";
+
+export const DESIGN_REVIEW_EXECUTION_REQUEST_SCHEMA = "pipeline.design-review-execution-request.proposed.v1";
+const sha = value => createHash("sha256").update(value).digest("hex");
+const hash = value => typeof value === "string" && /^[a-f0-9]{64}$/u.test(value);
+const oid = value => typeof value === "string" && /^[a-f0-9]{40}$/u.test(value);
+const keys = (value, expected) => value !== null && typeof value === "object" && !Array.isArray(value)
+  && Object.keys(value).length === expected.length && expected.every(key => Object.hasOwn(value, key));
+const pathOk = value => typeof value === "string" && value.length > 0 && value.length <= 240 && value === value.trim()
+  && !/[\\\x00-\x1f:]/u.test(value) && !value.split("/").some(part => !part || part === "." || part === "..");
+const refuse = code => ({ ok: false, code });
+const copy = value => structuredClone(value);
+const sourceKeys = ["path", "sha256", "blobOid", "mode"];
+const requestKeys = ["schema", "status", "featureId", "candidate", "sources", "specPath", "criteria", "authorExecutionSha256", "reviewerDuty", "round", "previousReceiptSha256", "assurance"];
+const assurance = Object.freeze({ class: "proposed-preparation", native: false, independentCriticPass: false });
+
+function validateRequest(request) {
+  if (!keys(request, requestKeys) || request.schema !== DESIGN_REVIEW_EXECUTION_REQUEST_SCHEMA || request.status !== "proposed"
+    || !/^[a-z][a-z0-9-]{0,63}$/u.test(request.featureId ?? "")
+    || !keys(request.candidate, ["commit", "tree"]) || !oid(request.candidate.commit) || !/^(?:[a-f0-9]{40}|[a-f0-9]{64})$/u.test(request.candidate.tree ?? "")
+    || !Array.isArray(request.sources) || request.sources.length < 2 || request.sources.length > 66
+    || !hash(request.authorExecutionSha256) || request.reviewerDuty !== "critic"
+    || ![1, 2].includes(request.round) || (request.round === 1 ? request.previousReceiptSha256 !== null : !hash(request.previousReceiptSha256))
+    || canonical(request.assurance) !== canonical(assurance)) return refuse("DRE-REQUEST-SHAPE");
+  for (let index = 0; index < request.sources.length; index++) {
+    const source = request.sources[index];
+    if (!keys(source, sourceKeys) || !pathOk(source.path) || !hash(source.sha256)
+      || !/^(?:[a-f0-9]{40}|[a-f0-9]{64})$/u.test(source.blobOid ?? "") || !["100644", "100755"].includes(source.mode)
+      || index > 0 && request.sources[index - 1].path >= source.path) return refuse("DRE-SOURCES");
+  }
+  const spec = request.sources.find(source => source.path === request.specPath);
+  if (!spec || !keys(request.criteria, ["specSha256", "ids", "sha256"]) || request.criteria.specSha256 !== spec.sha256
+    || !Array.isArray(request.criteria.ids) || !request.criteria.ids.length || request.criteria.ids.length > 4096
+    || request.criteria.ids.some((id, index) => typeof id !== "string" || !/^AC-(?:[1-9][0-9]*|E[1-9][0-9]*-[1-9][0-9]*-[0-9]{2})$/u.test(id)
+      || index > 0 && request.criteria.ids[index - 1] >= id)
+    || request.criteria.sha256 !== sha(canonical({ specSha256: spec.sha256, ids: request.criteria.ids }))) return refuse("DRE-CRITERIA");
+  return { ok: true, requestSha256: sha(canonical(request)) };
+}
+
+export function validateDesignReviewExecutionRequest(request) {
+  try { return validateRequest(request); } catch { return refuse("DRE-REQUEST-SHAPE"); }
+}
+
+/** Inputs are observations, not authenticated identities. The disk preparer obtains source bytes itself. */
+export function prepareDesignReviewExecution(input) {
+  if (!keys(input, ["featureId", "candidate", "sources", "specPath", "specBytes", "authorExecutionSha256", "round", "previousReceiptSha256"])
+    || !(input.specBytes instanceof Uint8Array)) return refuse("DRE-PREPARE-SHAPE");
+  let parsed;
+  try { parsed = inspectSpecCriterionRows(new TextDecoder("utf-8", { fatal: true }).decode(input.specBytes)); }
+  catch { return refuse("DRE-SPEC-BYTES"); }
+  if (!parsed.ok || !parsed.located.length) return refuse("DRE-CRITERIA");
+  const ids = [...parsed.located].sort();
+  const specSha256 = sha(input.specBytes);
+  const request = { schema: DESIGN_REVIEW_EXECUTION_REQUEST_SCHEMA, status: "proposed", featureId: input.featureId,
+    candidate: input.candidate, sources: input.sources, specPath: input.specPath,
+    criteria: { specSha256, ids, sha256: sha(canonical({ specSha256, ids })) },
+    authorExecutionSha256: input.authorExecutionSha256, reviewerDuty: "critic", round: input.round,
+    previousReceiptSha256: input.previousReceiptSha256, assurance: copy(assurance) };
+  const checked = validateDesignReviewExecutionRequest(request);
+  return checked.ok ? { ...checked, request: copy(request) } : checked;
+}
+
+/** Pure compare; cannot consume a review, change State or grant independent assurance. */
+export function recheckDesignReviewExecutionRequest({ request, current } = {}) {
+  const original = validateDesignReviewExecutionRequest(request), observed = validateDesignReviewExecutionRequest(current);
+  if (!original.ok) return original;
+  if (!observed.ok) return observed;
+  return original.requestSha256 === observed.requestSha256 ? { ok: true, status: "proposed", independentCriticPass: false }
+    : refuse("DRE-PREPARATION-DRIFT");
+}

+#!/usr/bin/env node
+// SPDX-License-Identifier: SUL-1.0
+/** Inert preparation only. This command never launches a reviewer or writes a receipt. */
+import { execFileSync } from "node:child_process";
+import { createHash } from "node:crypto";
+import { lstatSync, readFileSync, realpathSync } from "node:fs";
+import { join, resolve } from "node:path";
+import { fileURLToPath } from "node:url";
+import { prepareDesignReviewExecution } from "../lib/design-review-execution-producer.mjs";
+
+const refuse = code => ({ ok: false, code });
+export function prepareDesignReviewExecutionFromDisk(input) {
+  try {
+    const root = realpathSync(input.root);
+    const git = args => execFileSync("git", args, { cwd: root, encoding: "utf8", maxBuffer: 1048576 }).trim();
+    const candidate = { commit: git(["rev-parse", "HEAD"]), tree: git(["rev-parse", "HEAD^{tree}"]) };
+    if (!Array.isArray(input.sourcePaths) || input.sourcePaths.length < 2 || input.sourcePaths.length > 66) return refuse("DRE-SOURCES");
+    const sourcePaths = [...input.sourcePaths].sort();
+    if (new Set(sourcePaths).size !== sourcePaths.length || !sourcePaths.includes(input.specPath)) return refuse("DRE-SOURCES");
+    const records = [];
+    let specBytes;
+    for (const path of sourcePaths) {
+      if (typeof path !== "string" || path.length > 240 || /[\\\x00-\x1f:]/u.test(path) || path !== path.trim()
+        || path.split("/").some(part => !part || part === "." || part === "..")) return refuse("DRE-SOURCE-PATH");
+      let current = root;
+      for (const part of path.split("/")) {
+        current = join(current, part);
+        if (lstatSync(current).isSymbolicLink()) return refuse("DRE-SOURCE-PATH");
+      }
+      if (!lstatSync(current).isFile() || lstatSync(current).size > 1048576) return refuse("DRE-SOURCE-PATH");
+      const entry = git(["ls-tree", "HEAD", "--", path]).match(/^(100644|100755) blob ([a-f0-9]{40}|[a-f0-9]{64})\t(.+)$/u);
+      if (!entry || entry[3] !== path) return refuse("DRE-SOURCE-UNTRACKED");
+      const bytes = readFileSync(current);
+      const committed = execFileSync("git", ["cat-file", "blob", entry[2]], { cwd: root, maxBuffer: 1048576 });
+      if (!bytes.equals(committed)) return refuse("DRE-SOURCE-DRIFT");
+      records.push({ path, sha256: createHash("sha256").update(bytes).digest("hex"), blobOid: entry[2], mode: entry[1] });
+      if (path === input.specPath) specBytes = bytes;
+    }
+    if (candidate.commit !== git(["rev-parse", "HEAD"]) || candidate.tree !== git(["rev-parse", "HEAD^{tree}"])) return refuse("DRE-CANDIDATE-DRIFT");
+    return prepareDesignReviewExecution({ featureId: input.featureId, candidate, sources: records, specPath: input.specPath,
+      specBytes, authorExecutionSha256: input.authorExecutionSha256, round: input.round, previousReceiptSha256: input.previousReceiptSha256 });
+  } catch { return refuse("DRE-SOURCE-UNAVAILABLE"); }
+}
+
+export function main(argv = process.argv.slice(2)) {
+  if (argv.length !== 2 || argv[0] !== "prepare-proposal") { process.stdout.write(`${JSON.stringify(refuse("DRE-PROPOSED-ONLY"))}\n`); return 2; }
+  let result;
+  try { result = prepareDesignReviewExecutionFromDisk(JSON.parse(readFileSync(argv[1], "utf8"))); }
+  catch { result = refuse("DRE-PREPARE-SHAPE"); }
+  process.stdout.write(`${JSON.stringify(result)}\n`);
+  return result.ok ? 0 : 2;
+}
