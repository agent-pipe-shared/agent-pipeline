+## Serial count pass
+
+`node scripts/sitting-count-pass.mjs --list <suites.json> --out <evidence.md> [--root <dir>] [--only <id>...] [--resume] [--timeout-ms N]` runs every suite of the list alone, one after the other, with a private TMPDIR under `<home>/pipeline-test-tmp/count-pass-<run>/<n>` (mode 0700, outside every repository), `NODE_TEST_CONTEXT` removed and a per-suite cap (default 30 minutes; a timed-out child is killed with its process group). An entry is `{ id, items, command, expect, heavy, why }`: `command` is the argument list after `node` (`["--test", "<file>"]`), `command: null` with an `unresolved` reason marks an item with no identified suite (listed in the evidence, never run), `expect.minPass` makes a run GREEN only at or above that many passes, and `heavy` is a note for the operator (run those last, alone). The shipped list is `specs/sprint-alfred-epic/plans/count-pass-suites.json`; its page `count-pass-suites.md` names the heavy and the unresolved entries.
+
+Results are GREEN, RED or TIMEOUT, with tests, pass, fail, skip and todo copied from the node `ℹ` summary lines (or the `N/N cases passed` / `N passed, N failed` / `project-onboarding-v3: N cases across ... A failure that is a killed child or a `null !== 0` process failure is retried once and marked `(retried)`; a plain red and a timeout are not. Without `--resume` an existing state file is refused, never overwritten. +// Default spawn: serial child in its own process group so a timeout also stops the grandchildren that `node --test` starts.
+export function defaultSpawn(command, args, options) {
+  return new Promise((resolvePromise) => {
+    const detached = process.platform !== "win32";
+    const child = nodeSpawn(command, args, { cwd: options.cwd, env: options.env, detached, stdio: ["ignore", "pipe", "pipe"] });
+    let chunks = [];
+    let size = 0;
+    let timedOut = false;
+    const take = (c) => {
+      chunks.push(c);
+      size += c.length;
+      while (size > OUTPUT_CAP && chunks.length > 1) size -= chunks.shift().length;
+    };
+    child.stdout.on("data", take);
+    child.stderr.on("data", take);
+    const killTree = () => {
+      try {
+        if (detached) process.kill(-child.pid, "SIGKILL");
+        else child.kill("SIGKILL");
+      } catch {
+        try { child.kill("SIGKILL"); } catch { /* already gone */ }
+      }
+    };
+    const timer = setTimeout(() => { timedOut = true; killTree(); }, options.timeout);
+    child.on("error", (error) => { clearTimeout(timer); resolvePromise({ status: null, signal: null, error, stdout: Buffer.concat(chunks).toString("utf8"), stderr: "" }); });
+    child.on("close", (status, signal) => {
+      clearTimeout(timer);
+      if (detached) { try { process.kill(-child.pid, "SIGKILL"); } catch { /* group already empty */ } }
+      const result = { status, signal, stdout: Buffer.concat(chunks).toString("utf8"), stderr: "" };
+      if (timedOut) result.error = Object.assign(new Error("timed out"), { code: "ETIMEDOUT" });
+      resolvePromise(result);
+    });
+  });

+// Counts from the node test runner summary lines; falls back to the two custom summaries some suites print.
+export function parseCounts(out) {
+  const num = (k) => {
+    const m = new RegExp(`^ℹ ${k} (\\d+)`, "m").exec(out);
+    return m ? Number(m[1]) : null;
+  };
+  let counts = { tests: num("tests"), pass: num("pass"), fail: num("fail"), skipped: num("skipped"), todo: num("todo"), custom: false };
+  if (counts.tests === null) {
+    const slash = /(\d+)\/(\d+) cases passed/.exec(out);
+    const word = /(\d+) passed, (\d+) failed/.exec(out);
+    const shards = /project-onboarding-v3: (\d+) cases across .*?, (\d+) shards failed/.exec(out);
+    if (shards) {
+      const declared = Number(shards[1]);
+      const failedShards = Number(shards[2]);
+      counts = { tests: declared, pass: failedShards === 0 ? declared : (out.match(/^PASS /gm) ?? []).length, fail: failedShards, skipped: 0, todo: 0, custom: true };
+    } else if (slash) counts = { tests: Number(slash[2]), pass: Number(slash[1]), fail: Number(slash[2]) - Number(slash[1]), skipped: 0, todo: 0, custom: true };
+    else if (word) counts = { tests: Number(word[1]) + Number(word[2]), pass: Number(word[1]), fail: Number(word[2]), skipped: 0, todo: 0, custom: true };
+  }
+  return counts;

+// Evidence must hold repository-relative text only: scrub the root, the home directory, the user name and any /home/<x> or C:\Users\<x> shape.
+export function makeScrubber({ root, home, user }) {
+  const literals = [resolve(root), home, user].filter((s) => typeof s === "string" && s.length > 2).sort((a, b) => b.length - a.length);
+  const esc = (s) => s.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

+  if (entry.expect && typeof entry.expect.minPass === "number" && (counts.pass ?? 0) < entry.expect.minPass) return "RED";

+async function runOne({ entry, index, root, tmpBase, timeoutMs, spawn, execPath, log }) {
+  const tmp = join(tmpBase, String(index));
+  const prepare = () => {
+    rmSync(tmp, { recursive: true, force: true });
+    mkdirSync(tmp, { recursive: true, mode: 0o700 });
+    chmodSync(tmp, 0o700);
+  };
+  const attempt = async () => {
+    prepare();
+    const env = { ...process.env, TMPDIR: tmp, TMP: tmp, TEMP: tmp };
+    delete env.NODE_TEST_CONTEXT;
+    const t0 = Date.now();
+    const r = await spawn(execPath, entry.command, { cwd: root, env, timeout: timeoutMs });
+    const secs = Math.round((Date.now() - t0) / 1000);
+    const out = `${r.stdout ?? ""}${r.stderr ?? ""}`;
+    const timedOut = !!(r.error && r.error.code === "ETIMEDOUT");
+    const counts = parseCounts(out);
+    return { r, out, secs, timedOut, counts, result: classify(r, counts, entry, timedOut) };
+  };
+  let a = await attempt();
+  let retried = false;
+  let firstAttempt = null;
+  if (a.result !== "GREEN" && isRetryable(a.r, a.out)) {
+    retried = true;
+    firstAttempt = `${a.result} (${a.r.signal ? `killed by ${a.r.signal}` : a.r.status === null ? "child killed" : `exit ${a.r.status}, null !== 0`})`;
+    log(`   retrying once: ${entry.id} first attempt ${firstAttempt}`);
+    a = await attempt();
+  }
+  if (a.result === "GREEN") rmSync(tmp, { recursive: true, force: true });
+  return {
+    id: entry.id, result: a.result, tests: a.counts.tests, pass: a.counts.pass, fail: a.counts.fail, skipped: a.counts.skipped, todo: a.counts.todo,
+    custom: a.counts.custom, seconds: a.secs, exit: a.r.status, signal: a.r.signal ?? null, retried, firstAttempt,
+    failing: a.result === "GREEN" ? [] : failingCases(a.out).slice(0, 8),
+    tail: a.counts.tests === null && a.result !== "GREEN" ? a.out.trim().split("\n").slice(-4).join(" | ").slice(0, 400) : "",
+    at: new Date().toISOString(),
+  };

+function readList(path) {
+  const data = JSON.parse(readFileSync(path, "utf8"));
+  const suites = Array.isArray(data) ? data : data.suites;
+  if (!Array.isArray(suites)) throw new Error(`${path}: expected an array or { "suites": [...] }`);
+  const seen = new Set();
+  for (const s of suites) {
+    if (typeof s.id !== "string" || s.id === "") throw new Error(`${path}: an entry has no id`);
+    if (seen.has(s.id)) throw new Error(`${path}: duplicate id ${s.id}`);
+    seen.add(s.id);
+    if (s.command !== null && s.command !== undefined && (!Array.isArray(s.command) || s.command.length === 0)) throw new Error(`${path}: ${s.id}: command must be a non-empty array or null`);
+  }
+  return suites;

+// Usage:   node --test scripts/sitting-count-pass.test.mjs
+import assert from "node:assert/strict";
+import { mkdtempSync, mkdirSync, readFileSync, rmSync, writeFileSync, existsSync } from "node:fs";
+import { tmpdir } from "node:os";
+import { join } from "node:path";
+import { after, before, describe, test } from "node:test";
+import { runCountPass, defaultSpawn, parseCounts, makeScrubber } from "./sitting-count-pass.mjs";
+
+let base;
+let root;
+const quiet = () => {};
+
+before(() => {
+  base = mkdtempSync(join(tmpdir(), "count-pass-test-"));
+  root = join(base, "tree");
+  mkdirSync(root, { recursive: true });
+  const t = (name, body) => writeFileSync(join(root, name), `import { test } from "node:test";\n${body}\n`);
+  t("green.test.mjs", 'test("one", () => {});\ntest("two", () => {});');
+  t("red.test.mjs", 'test("good case", () => {});\ntest("broken case alpha", () => { throw new Error("boom"); });');
+  t("todo.test.mjs", 'test("ok", () => {});\ntest("later", { todo: true }, () => { throw new Error("not yet"); });\ntest("typed skip", { skip: "windows only" }, () => {});');
+  t("slow.test.mjs", 'test("never ends", async () => { await new Promise((r) => setTimeout(r, 60000)); });');
+});
+
+after(() => rmSync(base, { recursive: true, force: true }));
+
+const writeList = (name, suites) => {
+  const p = join(base, name);
+  writeFileSync(p, JSON.stringify({ suites }));
+  return p;
+};
+const entry = (id, file, extra = {}) => ({ id, items: [`pipeline.${id}`], command: ["--test", file], expect: null, heavy: false, why: "fixture", ...extra });
+
+describe("sitting-count-pass", () => {
+  test("SCP01 counts green, red and todo/skip suites and writes repository-relative evidence", async () => {
+    const list = writeList("l1.json", [entry("green", "green.test.mjs"), entry("red", "red.test.mjs"), entry("todo", "todo.test.mjs"), { id: "nosuite", items: [], command: null, unresolved: "no suite found for the fixture", why: "fixture" }]);
+    const out = join(base, "e1.md");
+    const lines = [];
+    const res = await runCountPass({ list, out, root, home: join(base, "home1"), log: (l) => lines.push(l) });
+    assert.equal(res.ok, false);
+    const byId = Object.fromEntries(res.records.map((r) => [r.id, r]));
+    assert.equal(byId.green.result, "GREEN");
+    assert.deepEqual([byId.green.tests, byId.green.pass, byId.green.fail], [2, 2, 0]);
+    assert.equal(byId.red.result, "RED");
+    assert.equal(byId.red.fail, 1);
+    assert.ok(byId.red.failing.some((n) => n.includes("broken case alpha")));
+    assert.equal(byId.todo.result, "GREEN");
+    assert.equal(byId.todo.todo, 1);
+    assert.equal(byId.todo.skipped, 1);
+    assert.equal(lines.filter((l) => /^(GREEN|RED|TIMEOUT) /.test(l)).length, 3, "one line per suite");
+    const md = readFileSync(out, "utf8");
+    assert.match(md, /## Red \(1\)/);
+    assert.match(md, /broken case alpha/);
+    assert.match(md, /## Skipped typed[\s\S]*`todo`/);
+    assert.match(md, /## Unresolved \(not run\)[\s\S]*`nosuite`/);
+    assert.match(md, /\| Id \| File \| Tests \| Pass \| Fail \| Skip \| Todo \| Seconds \| Result \|/);
+    assert.ok(!md.includes(root), "no absolute root path in the evidence");
+    assert.ok(!md.includes(base), "no absolute temp path in the evidence");
+    assert.ok(!/\/(home|Users)\//.test(md), "no home path in the evidence");
+    assert.ok(existsSync(`${out}.state.jsonl`));
+  });
+
+  test("SCP02 a suite that exceeds the cap is TIMEOUT and its process group is stopped", async () => {
+    const list = writeList("l2.json", [entry("slow", "slow.test.mjs")]);
+    const out = join(base, "e2.md");
+    const t0 = Date.now();
+    const res = await runCountPass({ list, out, root, home: join(base, "home2"), timeoutMs: 2000, log: quiet });
+    assert.equal(res.records[0].result, "TIMEOUT");
+    assert.equal(res.records[0].retried, false, "a timeout is not retried");
+    assert.ok(Date.now() - t0 < 30000, "the child was stopped near the cap");
+    assert.match(readFileSync(out, "utf8"), /TIMEOUT/);
+  });
+
+  test("SCP03 --resume skips a finished suite and a second run without it is refused", async () => {
+    const list = writeList("l3.json", [entry("green", "green.test.mjs"), entry("todo", "todo.test.mjs")]);
+    const out = join(base, "e3.md");
+    await runCountPass({ list, out, root, home: join(base, "home3"), only: ["green"], log: quiet });
+    const spawned = [];
+    const spy = (cmd, args, opts) => { spawned.push(args.join(" ")); return defaultSpawn(cmd, args, opts); };
+    await assert.rejects(() => runCountPass({ list, out, root, home: join(base, "home3"), spawn: spy, log: quiet }), /--resume/);
+    assert.equal(spawned.length, 0);
+    const lines = [];
+    const res = await runCountPass({ list, out, root, home: join(base, "home3"), resume: true, spawn: spy, log: (l) => lines.push(l) });
+    assert.deepEqual(spawned, ["--test todo.test.mjs"], "only the unfinished suite ran");
+    assert.ok(lines.some((l) => l.startsWith("SKIP-DONE green")));
+    assert.equal(res.records.length, 2);
+    assert.match(readFileSync(out, "utf8"), /2 suites run/);
+  });
+
+  test("SCP04 a null !== 0 process failure is retried once and recorded as retried", async () => {
+    const list = writeList("l4.json", [entry("flaky", "green.test.mjs")]);
+    const calls = [];
+    const spawn = async (_cmd, _args, opts) => {
+      calls.push(opts.env.TMPDIR);
+      if (calls.length === 1) return { status: 1, signal: null, stdout: "not ok 1\n  Process failed: null !== 0\nℹ tests 1\nℹ pass 0\nℹ fail 1\nℹ skipped 0\nℹ todo 0\n", stderr: "" };
+      return { status: 0, signal: null, stdout: "ℹ tests 1\nℹ pass 1\nℹ fail 0\nℹ skipped 0\nℹ todo 0\n", stderr: "" };
+    };
+    const out = join(base, "e4.md");
+    const res = await runCountPass({ list, out, root, home: join(base, "home4"), spawn, log: quiet });
+    assert.equal(calls.length, 2);
+    assert.equal(res.records[0].result, "GREEN");
+    assert.equal(res.records[0].retried, true);
+    assert.match(res.records[0].firstAttempt, /null !== 0/);
+    assert.match(readFileSync(out, "utf8"), /GREEN \(retried\)/);
+    assert.ok(calls[0].includes("pipeline-test-tmp"), "private TMPDIR under the injected home");
+  });
+
+  test("SCP05 a killed child is retried once, a plain red is not, and the child env has no NODE_TEST_CONTEXT", async () => {
+    const list = writeList("l5.json", [entry("killed", "green.test.mjs"), entry("plain", "red.test.mjs")]);
+    let killedCalls = 0;
+    let plainCalls = 0;
+    let sawContext = false;
+    const prev = process.env.NODE_TEST_CONTEXT;
+    process.env.NODE_TEST_CONTEXT = "child-v8";
+    const spawn = async (_cmd, args, opts) => {
+      if ("NODE_TEST_CONTEXT" in opts.env) sawContext = true;
+      if (args.includes("green.test.mjs")) { killedCalls += 1; return killedCalls === 1 ? { status: null, signal: "SIGKILL", stdout: "", stderr: "" } : { status: 0, signal: null, stdout: "ℹ tests 2\nℹ pass 2\nℹ fail 0\nℹ skipped 0\nℹ todo 0\n", stderr: "" }; }
+      plainCalls += 1;
+      return { status: 1, signal: null, stdout: "ℹ tests 2\nℹ pass 1\nℹ fail 1\nℹ skipped 0\nℹ todo 0\n", stderr: "" };
+    };
+    try {
+      const res = await runCountPass({ list, out: join(base, "e5.md"), root, home: join(base, "home5"), spawn, log: quiet });
+      assert.equal(killedCalls, 2);
+      assert.equal(plainCalls, 1);
+      assert.equal(res.records[0].result, "GREEN");
+      assert.equal(res.records[1].result, "RED");
+      assert.equal(res.records[1].retried, false);
+      assert.equal(sawContext, false);
+    } finally {
+      if (prev === undefined) delete process.env.NODE_TEST_CONTEXT;
+      else process.env.NODE_TEST_CONTEXT = prev;
+    }
+  });
+
+  test("SCP06 expect.minPass turns a thin green into RED, and suites never overlap", async () => {
+    const list = writeList("l6.json", [entry("thin", "green.test.mjs", { expect: { minPass: 5 } }), entry("second", "todo.test.mjs")]);
+    let running = 0;
+    let maxRunning = 0;
+    const spawn = async (cmd, args, opts) => {
+      running += 1;
+      maxRunning = Math.max(maxRunning, running);
+      try { return await defaultSpawn(cmd, args, opts); } finally { running -= 1; }
+    };
+    const res = await runCountPass({ list, out: join(base, "e6.md"), root, home: join(base, "home6"), spawn, log: quiet });
+    assert.equal(res.records[0].result, "RED");
+    assert.equal(maxRunning, 1);
+  });
+
+  test("SCP07 only counts and path scrubbing helpers behave", () => {
+    assert.deepEqual(parseCounts("ℹ tests 3\nℹ pass 2\nℹ fail 0\nℹ skipped 1\nℹ todo 0\n"), { tests: 3, pass: 2, fail: 0, skipped: 1, todo: 0, custom: false });
+    assert.equal(parseCounts("161/161 cases passed").tests, 161);
+    assert.equal(parseCounts("24 passed, 0 failed").pass, 24);
+    assert.equal(parseCounts("nothing").tests, null);
+    const ctl = parseCounts("PASS  a\nproject-onboarding-v3: 216 cases across 3 sequential chunks of 4 shards, 0 shards failed\n");
+    assert.deepEqual([ctl.tests, ctl.pass, ctl.fail, ctl.custom], [216, 216, 0, true]);
+    const scrub = makeScrubber({ root: "/work/some/tree", home: "/home/alice", user: "alice" });
+    assert.equal(scrub("at /work/some/tree/x.mjs and /home/alice/y and /home/bob/z"), "at <path>/x.mjs and <path>/y and <path>/z");
+  });
+});

+# Count-pass suite list (K110)
+
+`count-pass-suites.json` is the input of `scripts/sitting-count-pass.mjs` (see `docs/operating-tools.md`, "Serial count pass"). It covers the 55 rows of `evidence/delivery-remeasure-2026-10-23/snapshot.json` that carry an open entry owned by `agent` or `verify-wsl`, plus the suites that the notes in `evidence/held-patches/` name as counted.
+
+120 entries: 69 serve at least one of the 55 rows (`items`), 47 are held-patch suites only (empty `items`, `why` names the notes), 4 are unresolved (`command: null`, never run). Entries are de-duplicated by suite file; the id is the file name without `.test.mjs`. Order: cheap suites first (by the seconds of an earlier counted run where one is on record, otherwise by name), the four heavy suites last, the unresolved entries at the end.
+
+Resolution rule for a row: its own pin `plugins/pipeline-core/lib/pins/<slug>.test.mjs` (all pins named by a row exist under the full slug), plus the suites its `open.what` or backlog text names. `why` carries the open sentence(s) verbatim, tagged `[agent]` or `[verify-wsl]`. Read `why` before trusting a green: for rows whose open clause asks to build, restate or document something, a counted pin run is the evidence gap only if that work is committed first.
