const render=argv=>[process.execPath,cli,...argv].map(value=>JSON.stringify(value)).join(' ');
function actualGuard(command,root,runner) {
  let stderr='',readinessCalls=0;
  const code=guardMain({tool_name:'Bash',tool_input:{command}}, {projectDir:root,runner,
    requireProjectOnboardingReadyFn:()=>{readinessCalls++;return {schema:'pipeline.project-onboarding-ready-gate.v1',status:'ready',intent:'session'};},
    writeErrorFn:value=>{stderr+=value;},recordHumanGuardDenialFn:()=>({status:'ineligible'})});
  return {code,stderr,readinessCalls};
}
const cases=[];
function test(id,name,run){cases.push({id,name,run});}

test('SHDQ001','actual Source help CLI and exact read-only admission',()=>{
  for(const flag of ['--help','-h']) {
    const command=[process.execPath,help,flag].map(value=>JSON.stringify(value)).join(' ');
    assert.equal(isSanctionedLifecycleCommand(command,source),true);
    const result=spawnSync(process.execPath,[help,flag],{cwd:source,encoding:'utf8',timeout:20000});
    assert.equal(result.status,0,result.stderr);assert.match(result.stdout,/Usage: po-human-approval.mjs/);
    assert.match(result.stdout,/sign-intent/);assert.match(result.stdout,/verify/);
  }
  assert.equal(isSanctionedLifecycleCommand([process.execPath,help,'--help','approve'].map(JSON.stringify).join(' '),source),false);
});

for(const scenario of ['valid','invalid','foreign','stale','replay']) {
  test(`SHDQ00${cases.length+1}`,`actual owned signed-plan import: ${scenario}`,async()=>{
    const f=await owner.createStageStateOwner('presented');
    const argv=['approve-plan','--design-approval-request',f.requestPath];
    assert.equal(isSanctionedLifecycleCommand(render(argv),f.dir),true,'transport is agent work; writer verifies proof');
    assert.equal(isSanctionedLifecycleCommand(render(['discard-feature','--by','synthetic-po','--reason','synthetic discard']),f.dir),false,'discard remains outside agent admission');
    if(scenario==='valid')for(const runner of ['claude','codex','antigravity']) {
      const admission=actualGuard(render(argv),f.dir,runner);
      assert.equal(admission.code,0,admission.stderr);assert.equal(admission.readinessCalls,1);
      const helpCommand=[process.execPath,help,'--help'].map(value=>JSON.stringify(value)).join(' ');
      const helped=actualGuard(helpCommand,f.dir,runner);
      assert.equal(helped.code,0,helped.stderr);
      const discarded=actualGuard(render(['discard-feature','--by','synthetic-po','--reason','synthetic discard']),f.dir,runner);
      console.log(JSON.stringify({schema:'pipeline.signing-help-guard-case.v1',runner,plan:admission.code,help:helped.code,discard:discarded.code,sharedEvaluatorOnly:true}));
    }
    const proofPath=join(f.dir,f.requestPath.replace('request-','proof-'));
    const original=JSON.parse(readFileSync(proofPath));
    if(scenario==='invalid')writeFileSync(proofPath,JSON.stringify({...original,signatureBase64:Buffer.alloc(64).toString('base64')}));
    if(scenario==='foreign') {
      const pair=generateKeyPairSync('ed25519');
      writeFileSync(proofPath,JSON.stringify({...original,keyReference:'foreign-disposable-key',publicKey:pair.publicKey.export({type:'spki',format:'pem'}),signatureBase64:sign(null,Buffer.from(original.intentSha256),pair.privateKey).toString('base64')}));
    }
    if(scenario==='stale')writeFileSync(proofPath,JSON.stringify({...original,intentSha256:'0'.repeat(64)}));
    const before=owner.staged.rawState(f),member=owner.readMember(f);
    const call=await owner.invokeOwnedState(f,owner.actualRequest(f,argv));
    assert.equal(call.result.status,'completed',JSON.stringify(call.result));
    if(['invalid','foreign','stale'].includes(scenario)) {
      assert.equal(call.result.exitCode,2,call.text);assert.match(call.text,/DAB-|PROOF|SIGNATURE/);
      assert.deepEqual(owner.staged.rawState(f),before,'rejected proof cannot change State');
      owner.preserveOwnerIdentity(member,owner.readMember(f));
      assert.equal(owner.staged.state(f).planApproved,false);
    } else {
      owner.requireSuccessfulCommand(call);owner.assertActualStateReadback(f,call,member);
      const approved=owner.staged.state(f);
      assert.equal(approved.planApproved,true);assert.equal(approved.planApproval.schema,'pipeline.plan-approval.v9');
      assert.equal(approved.planApproval.designApproval.mode,'signature');
      assert.equal(approved.continuity.queueHead.dispatch,null,'proof import does not register work');
      assert.equal(approved.pushApproval,undefined,'plan approval cannot approve push');
      if(scenario==='replay') {
        const saved=owner.staged.rawState(f);
        const replay=await owner.invokeOwnedState(f,owner.actualRequest(f,argv));
        assert.equal(replay.result.status,'completed');assert.equal(replay.result.exitCode,2,replay.text);
        assert.deepEqual(owner.staged.rawState(f),saved,'proof replay records no second approval');
      }
    }
    console.log(JSON.stringify({schema:'pipeline.signing-help-qualification-case.v1',scenario,entry:argv[0],exit:call.result.exitCode,proof:'synthetic-owned-fixture',stateChanged:!owner.staged.rawState(f).equals(before),nativeAttestation:false}));
  });
}

  assert.equal(decision.code,2,'PO-only discard must remain refused after readiness');
});
