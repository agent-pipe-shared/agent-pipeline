// diagnostic: run a suite as Verify would (FD3 announced and writable)
import { spawnSync } from "node:child_process";
import { openSync, mkdirSync } from "node:fs";
import { homedir } from "node:os";
import { join } from "node:path";
const file = process.argv[2];
const tmp = join(homedir(), "pipeline-test-tmp", "g3fd");
mkdirSync(tmp, { recursive: true, mode: 0o700 });
const fd = openSync("/dev/null", "w");
const env = { ...process.env, PIPELINE_VERIFY_CASE_COMPLETION_FD: "3", PIPELINE_VERIFY_CASE_COMPLETION_MAX_BYTES: "65536", TMPDIR: tmp, TMP: tmp, TEMP: tmp };
delete env.NODE_TEST_CONTEXT;
const r = spawnSync(process.execPath, [file], { cwd: new URL(".", import.meta.url).pathname, env, stdio: ["ignore", "pipe", "pipe", fd], encoding: "utf8", timeout: 600000 });
const out = (r.stdout ?? "") + (r.stderr ?? "");
console.log("exit", r.status);
console.log(out.split("\n").filter((l) => /^(ℹ (tests|pass|fail))|✖|Error|not ok/.test(l)).slice(0, 40).join("\n"));
