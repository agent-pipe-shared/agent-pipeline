+check("push guard gets a bounded share sized above the recorded run while other guards retain defaults", () => {
+  const source = readFileSync(adapter, "utf8");
+  const outerBudget = Number(source.match(/const HOOK_BUDGET_MS = ([\d_]+);/u)?.[1].replaceAll("_", ""));
+  const push = source.match(/"guard-push\.mjs": \{ capMs: ([\d_]+), reserveMs: ([\d_]+) \}/u);
+  const defaults = source.match(/default: \{ capMs: ([\d_]+), reserveMs: ([\d_]+) \}/u);
+  assert.equal(outerBudget, 42_000);
+  assert.ok(push, "push guard must have its own explicit bounded share");
+  const [, pushCapText, pushReserveText] = push;
+  const pushCap = Number(pushCapText.replaceAll("_", ""));
+  const pushReserve = Number(pushReserveText.replaceAll("_", ""));
+  assert.ok(pushCap > 14_090, "push share must exceed the observed successful 14,090ms run");
+  assert.ok(Number.isFinite(pushCap) && pushCap < outerBudget);
+  assert.ok(Number.isFinite(pushReserve) && pushReserve >= 5_000);
+  assert.ok(pushCap + pushReserve < outerBudget, "push timeout must leave recovery headroom under the outer cap");
+  assert.deepEqual(defaults?.slice(1).map((value) => Number(value.replaceAll("_", ""))), [8_000, 5_000]);
+  assert.match(source, /if \(available <= 0\) return timedOutSpawnResult\(\);/u,
+    "outer exhaustion must remain a typed fail-closed timeout");
+
+  // Production boundedSpawn chooses min(cap, outer - elapsed - reserve).
+  // A preceding default guard consuming its full cap still leaves the push
+  // guard's full share and more than the declared reserve before outer timeout.
+  const afterPriorDefault = outerBudget - 8_000;
+  const available = afterPriorDefault - pushReserve;
+  const timeout = Math.max(1, Math.min(pushCap, available));
+  assert.equal(timeout, pushCap);
+  assert.ok(afterPriorDefault - timeout >= pushReserve);
+  const exhaustedElapsed = outerBudget - pushReserve;
+  assert.ok(outerBudget - exhaustedElapsed - pushReserve <= 0,
+    "an exhausted outer budget must deny before spawning another child");
+});
