+
+`node scripts/post-sitting-checks.mjs [--root <dir>] [--only <name>...] [--json <file>]` runs the light end-state checks of the sitting one after the other, each with a private TMPDIR outside every repository and `NODE_TEST_CONTEXT` removed, and prints one row per check (verdict, exit code, seconds, one-line result) and a total. The expected counts are constants at the top of the script; move them when a later commit changes them. `--json <file>` writes the table as JSON and is refused (exit 2) for a path outside `<root>/scratch/`. Heavy suites are not part of it: they belong to the serial count pass above. Before the sitting's protected steps several rows are RED by design (unregistered suites, reader-input drift).

+export function runChecks({ root = ".", only = [], home = homedir(), log = console.log, execPath = process.execPath } = {}) {
+  const absRoot = resolve(root);
+  for (const n of only) if (!CHECKS.some((c) => c.name === n)) throw Object.assign(new Error(`--only names a check that does not exist: ${n} (known: ${CHECKS.map((c) => c.name).join(", ")})`), { usage: true });

+  const tmpBase = join(home, "pipeline-test-tmp", `post-sitting-${process.pid}`);
+  const rows = [];
+  CHECKS.filter((c) => only.length === 0 || only.includes(c.name)).forEach((c, i) => {
+    const tmp = join(tmpBase, String(i));
+    rmSync(tmp, { recursive: true, force: true });
+    mkdirSync(tmp, { recursive: true, mode: 0o700 });
+    chmodSync(tmp, 0o700);

+    const out = `${r.stdout ?? ""}${r.stderr ?? ""}`;

+    if (verdict.ok) rmSync(tmp, { recursive: true, force: true });
+    const row = { check: c.name, command: `node ${args.join(" ")}`, verdict: verdict.ok ? "GREEN" : "RED", exit: r.status, seconds, result: scrub(verdict.line).slice(0, 200) };
+    rows.push(row);
+    log(`${row.verdict.padEnd(5)} ${c.name.padEnd(24)} exit ${String(row.exit).padEnd(4)} ${String(seconds).padStart(5)} s  ${row.result}`);

+  }
