// Test-only placement port for the two fixtures that require a real outside-Git root.
import { realpathSync } from "node:fs";
import { dirname, relative, resolve, sep } from "node:path";
import { fileURLToPath } from "node:url";
import { validateCreatedHome, validatePrivateParent } from "../scripts/codex-critic-private-parent-fixture.mjs";

const PRIVATE_PARENT = "/run/user/1000/ap-g17";
const PROJECT_ROOT = realpathSync(fileURLToPath(new URL("../../../", import.meta.url)));

export function originalGuardOutsideFixtureRoot() {
  const supplied = process.env.PIPELINE_GUARD_EXTERNAL_FIXTURE_ROOT;
  if (supplied === undefined) return null;
  if (!supplied || resolve(supplied) !== supplied || realpathSync(supplied) !== supplied ||
      dirname(supplied) !== PRIVATE_PARENT) {
    throw new Error("ORIGINAL-GUARD-FIXTURE-ROOT: expected a direct, normalized, physical private-parent child");
  }
  validatePrivateParent(PRIVATE_PARENT);
  validateCreatedHome(supplied);
  const fromProject = relative(PROJECT_ROOT, supplied);
  if (fromProject === "" || (fromProject !== ".." && !fromProject.startsWith(`..${sep}`))) {
    throw new Error("ORIGINAL-GUARD-FIXTURE-ROOT: outside-Git fixture is inside project root");
  }
  return supplied;
}
