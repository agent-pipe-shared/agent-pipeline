import { createHash, randomUUID } from "node:crypto";
import { execFileSync } from "node:child_process";
import { performance } from "node:perf_hooks";
import {
  lstatSync,
  linkSync,
  mkdirSync,
  readFileSync,
  realpathSync,
  renameSync,
  unlinkSync,
  writeFileSync,
} from "node:fs";
import { dirname, join, resolve } from "node:path";

import { prepareRoleDispatchBatch } from "./role-dispatch-preflight.mjs";
import { evaluateDispatchModel } from "./dispatch-model-enforcement.mjs";
import { isShippedPipelineAgentType } from "./dispatch-policy.mjs";

export const AGY_NATIVE_BATCH_ARTIFACT_SCHEMA = "pipeline.antigravity-native-dispatch-batch.v1";
export const AGY_NATIVE_BATCH_PREPARATION_SCHEMA = "pipeline.antigravity-native-dispatch-preparation.v1";
export const AGY_NATIVE_BATCH_VERDICT_SCHEMA = "pipeline.antigravity-native-dispatch-verdict.v1";

const SHA256 = /^[a-f0-9]{64}$/u;
const OID = /^(?:[a-f0-9]{40}|[a-f0-9]{64})$/u;
const MAX_ARTIFACT_BYTES = 1024 * 1024;
const DEFAULT_TTL_MS = 10 * 60 * 1000;
// Keep subprocess teardown overhead inside the externally visible five-second PREPARE bound.
const PREPARE_DEADLINE_MS = 4_500;
const ARTIFACT_KEYS = ["artifactSha256", "callShape", "candidate", "expiresAtEpochMs", "nativeSubagents", "packets", "repositoryRootSha256", "schema"];
const CRITIC_REVIEW_KEYS = ["authorId", "baseCommit", "diffSha256", "packetSha256", "reviewedPaths"];
const verifiedCriticContexts = new WeakMap();
const observedCriticReturns = new WeakMap();
const inspectedCriticReturns = new WeakMap();
const ENTRY_KEYS = new Set(["Model", "Prompt", "Role", "TypeName", "Workspace"]);
const NATIVE_WORKSPACES = new Set(["inherit", "branch", "share"]);

const sha256 = (value) => createHash("sha256").update(value).digest("hex");
const exactKeys = (value, keys) => value !== null && typeof value === "object" && !Array.isArray(value)
  && JSON.stringify(Object.keys(value).sort()) === JSON.stringify([...keys].sort());

function rejected(code, field = null) {
  return { schema: AGY_NATIVE_BATCH_VERDICT_SCHEMA, status: "rejected", code, field, modelCalls: 0, launcherCalls: 0 };
}

function git(root, args, deadlineEpochMs) {
  const remainingMs = Math.floor(deadlineEpochMs - performance.now());
  if (remainingMs < 1) return null;
  try {
    return String(execFileSync("git", args, {
      cwd: root,
      encoding: "utf8",
      shell: false,
      stdio: ["ignore", "pipe", "ignore"],
      timeout: remainingMs,
      maxBuffer: 1024 * 1024,
    })).trim();
  } catch { return null; }
}

function gitBytes(root, args, deadlineEpochMs) {
  const remainingMs = Math.floor(deadlineEpochMs - performance.now());
  if (remainingMs < 1) return null;
  try { return execFileSync("git", args, { cwd: root, shell: false, stdio: ["ignore", "pipe", "ignore"], timeout: remainingMs, maxBuffer: 4 * 1024 * 1024 }); }
  catch { return null; }
}

function criticPacketIndex(packets) {
  const indexes = packets.flatMap((packet, index) => bareRole(packet.role) === "critic" ? [index] : []);
  return indexes.length === 1 ? indexes[0] : -1;
}

const CHILD_ID = /^[A-Za-z0-9][A-Za-z0-9._:-]{0,127}$/u;

function reviewFacts(root, candidate, packets, requested, deadlineEpochMs) {
  const index = criticPacketIndex(packets);
  const candidateCommit = candidate.commit ?? candidate.head;
  if (index < 0 || !exactKeys(requested, ["authorId", "baseCommit", "reviewedPaths"]) || !CHILD_ID.test(requested.authorId ?? "") || !OID.test(requested.baseCommit ?? "")
    || !Array.isArray(requested.reviewedPaths) || requested.reviewedPaths.length < 1 || requested.reviewedPaths.length > 128
    || requested.reviewedPaths.some(path => typeof path !== "string" || !path || path.startsWith("/") || path.includes("\\") || path.split("/").some(part => !part || part === "." || part === ".."))) return null;
  const paths = [...new Set(requested.reviewedPaths)].sort();
  if (paths.length !== requested.reviewedPaths.length) return null;
  const baseTree = git(root, ["rev-parse", "--verify", `${requested.baseCommit}^{tree}`], deadlineEpochMs);
  const ancestry = git(root, ["merge-base", requested.baseCommit, candidateCommit], deadlineEpochMs);
  if (!OID.test(baseTree ?? "") || ancestry !== requested.baseCommit) return null;
  const diff = gitBytes(root, ["diff", "--binary", requested.baseCommit, candidateCommit, "--", ...paths], deadlineEpochMs);
  if (!diff || !diff.length) return null;
  return { authorId: requested.authorId, baseCommit: requested.baseCommit, diffSha256: sha256(diff), packetSha256: sha256(JSON.stringify(packets[index])), reviewedPaths: paths };
}

function reviewWorktreeClean(root, paths, deadlineEpochMs) {
  const status = gitBytes(root, ["status", "--porcelain=v1", "-z", "--untracked-files=all", "--", ...paths], deadlineEpochMs);
  return status !== null && status.length === 0;
}

function repositoryState(root, deadlineEpochMs = performance.now() + PREPARE_DEADLINE_MS) {
  let physicalRoot;
  try { physicalRoot = realpathSync(resolve(root)); }
  catch { return null; }
  const commonRaw = git(physicalRoot, ["rev-parse", "--path-format=absolute", "--git-common-dir"], deadlineEpochMs);
  const head = git(physicalRoot, ["rev-parse", "--verify", "HEAD"], deadlineEpochMs);
  const tree = head === null ? null : git(physicalRoot, ["rev-parse", `${head}^{tree}`], deadlineEpochMs);
  if (commonRaw === null || !OID.test(head ?? "") || !OID.test(tree ?? "")) return null;
  try {
    const commonDir = realpathSync(resolve(commonRaw));
    return { physicalRoot, commonDir, head, tree };
  } catch { return null; }
}

function validNativeSubagents(value) {
  return Array.isArray(value) && value.length > 0 && value.length <= 64
    && value.every((entry) => entry !== null && typeof entry === "object" && !Array.isArray(entry)
      && Object.keys(entry).every((key) => ENTRY_KEYS.has(key))
      && ["TypeName", "Role", "Prompt"].every((key) => Object.hasOwn(entry, key))
      && typeof entry.TypeName === "string" && entry.TypeName.trim() === entry.TypeName && entry.TypeName !== ""
      && typeof entry.Role === "string" && entry.Role.trim() === entry.Role && entry.Role !== ""
      && typeof entry.Prompt === "string" && entry.Prompt.trim() !== ""
      && (!Object.hasOwn(entry, "Model") || typeof entry.Model === "string")
      && (!Object.hasOwn(entry, "Workspace") || (typeof entry.Workspace === "string" && NATIVE_WORKSPACES.has(entry.Workspace))));
}

function bareRole(value) {
  return typeof value === "string" && value.startsWith("pipeline-core:")
    ? value.slice("pipeline-core:".length)
    : value;
}

function nativeTierRefusal(nativeSubagents) {
  for (const entry of nativeSubagents) {
    const type = bareRole(entry.TypeName);
    if (!isShippedPipelineAgentType(type)) continue;
    const evaluation = evaluateDispatchModel({ runner: "antigravity", subagentType: type, model: entry.Model });
    if (!evaluation.ok) return evaluation.findings[0]?.code ?? "DISPATCH-MODEL-UNROUTED";
  }
  return null;
}

function entriesMatchPackets(nativeSubagents, packets) {
  return nativeSubagents.length === packets.length && nativeSubagents.every((entry, index) => (
    bareRole(entry.TypeName) === bareRole(packets[index]?.role)
    && entry.Prompt === packets[index]?.prompt
  ));
}

function artifactBody({ state, nativeSubagents, packets, expiresAtEpochMs, criticReview }) {
  return {
    schema: AGY_NATIVE_BATCH_ARTIFACT_SCHEMA,
    repositoryRootSha256: sha256(state.physicalRoot),
    candidate: { commit: state.head, tree: state.tree },
    expiresAtEpochMs,
    nativeSubagents: structuredClone(nativeSubagents),
    packets: structuredClone(packets),
    ...(criticReview ? { criticReview: structuredClone(criticReview) } : {}),
  };
}

function artifactDirectory(commonDir) {
  return join(commonDir, "agent-pipeline", "run", "antigravity-native-dispatch-batches");
}

function artifactPath(commonDir, nativeSubagents, physicalRoot, callShape = "batch") {
  return join(artifactDirectory(commonDir), `${sha256(physicalRoot)}-${sha256(JSON.stringify({ callShape, nativeSubagents }))}.json`);
}

function parseAuthenticatedRetirableArtifact(raw, { state, nativeSubagents, callShape, nowEpochMs }) {
  let artifact;
  try { artifact = JSON.parse(raw); }
  catch { return null; }
  if (!exactKeys(artifact, artifact?.criticReview ? [...ARTIFACT_KEYS, "criticReview"] : ARTIFACT_KEYS) || artifact.schema !== AGY_NATIVE_BATCH_ARTIFACT_SCHEMA
    || !SHA256.test(artifact.artifactSha256 ?? "") || !SHA256.test(artifact.repositoryRootSha256 ?? "")
    || !Number.isSafeInteger(artifact.expiresAtEpochMs) || !exactKeys(artifact.candidate, ["commit", "tree"])) return null;
  const { artifactSha256, ...body } = artifact;
  if (sha256(JSON.stringify(body)) !== artifactSha256
    || artifact.repositoryRootSha256 !== sha256(state.physicalRoot)
    || artifact.callShape !== callShape
    || JSON.stringify(artifact.nativeSubagents) !== JSON.stringify(nativeSubagents)) return null;
  return artifact.expiresAtEpochMs < nowEpochMs
    || artifact.candidate.commit !== state.head
    || artifact.candidate.tree !== state.tree
    ? artifactSha256
    : null;
}

function publishReplacingRetirable({ temporary, path, state, nativeSubagents, callShape, nowEpochMs }) {
  try {
    linkSync(temporary, path);
    return "published";
  } catch (error) {
    if (error?.code !== "EEXIST") return "write-failed";
  }

  let observedRaw;
  try { observedRaw = readFileSync(path, "utf8"); }
  catch { return "exists"; }
  const observedDigest = parseAuthenticatedRetirableArtifact(observedRaw, { state, nativeSubagents, callShape, nowEpochMs });
  if (observedDigest === null) return "exists";

  const retired = join(dirname(path), `.${randomUUID()}.retired`);
  let claimed = false;
  try {
    renameSync(path, retired);
    claimed = true;
    const claimedRaw = readFileSync(retired, "utf8");
    const claimedDigest = parseAuthenticatedRetirableArtifact(claimedRaw, { state, nativeSubagents, callShape, nowEpochMs });
    if (claimedRaw !== observedRaw || claimedDigest !== observedDigest) {
      try { linkSync(retired, path); } catch { /* another publisher owns the canonical name */ }
      return "exists";
    }
    try {
      linkSync(temporary, path);
      return "published";
    } catch {
      try { linkSync(retired, path); } catch { /* preserve the winner if the name was concurrently acquired */ }
      return "write-failed";
    }
  } catch {
    return "exists";
  } finally {
    if (claimed) {
      try { unlinkSync(retired); } catch { /* best-effort cleanup after canonical publication/restoration */ }
    }
  }
}

/**
 * Prepare, but never launch, one native Antigravity invoke_subagent batch.
 * The caller passes the returned Subagents array unchanged to the runner's built-in tool.
 */
export function prepareAntigravityNativeDispatch({ root, packets, nativeSubagents, callShape = "batch", criticReview, nowEpochMs = Date.now(), ttlMs = DEFAULT_TTL_MS } = {}) {
  const prepareDeadlineEpochMs = performance.now() + PREPARE_DEADLINE_MS;
  const state = repositoryState(root, prepareDeadlineEpochMs);
  if (state === null) return rejected("AGY-NATIVE-ROOT", "root");
  if (!validNativeSubagents(nativeSubagents)) return rejected("AGY-NATIVE-SUBAGENTS", "nativeSubagents");
  if (!["flat", "batch"].includes(callShape)) return rejected("AGY-NATIVE-CALL-SHAPE", "callShape");
  if (callShape === "flat" && nativeSubagents.length !== 1) return rejected("AGY-NATIVE-CALL-SHAPE", "nativeSubagents");
  const tierRefusal = nativeTierRefusal(nativeSubagents);
  if (tierRefusal !== null) return rejected(tierRefusal, "nativeSubagents.Model");
  if (!Number.isSafeInteger(nowEpochMs) || nowEpochMs < 0 || !Number.isSafeInteger(ttlMs) || ttlMs < 1_000 || ttlMs > DEFAULT_TTL_MS) {
    return rejected("AGY-NATIVE-TTL", "ttlMs");
  }
  const batch = prepareRoleDispatchBatch({
    root: state.physicalRoot,
    packets,
    deadlineEpochMs: prepareDeadlineEpochMs,
  });
  if (batch.status !== "prepared") return { ...rejected(batch.code, "packets"), preparation: batch };
  if (packets.some((packet) => packet.transport !== "antigravity" || packet.resultDestination?.kind !== "return")) {
    return rejected("AGY-NATIVE-PACKET-BOUNDARY", "packets");
  }
  if (packets.some((packet) => packet.candidate.commit !== state.head || packet.candidate.tree !== state.tree)) {
    return rejected("AGY-NATIVE-CANDIDATE", "packets.candidate");
  }
  if (!entriesMatchPackets(nativeSubagents, packets)) return rejected("AGY-NATIVE-ARRAY-MISMATCH", "nativeSubagents");

  const criticCount = packets.filter(packet => bareRole(packet.role) === "critic").length;
  if (criticCount > 1) return rejected("AGY-CRITIC-REVIEW-AMBIGUOUS", "packets");
  const criticIndex = criticPacketIndex(packets);
  if (criticIndex >= 0 && !criticReview) return rejected("AGY-CRITIC-REVIEW-REQUIRED", "criticReview");
  if (criticReview && criticIndex < 0) return rejected("AGY-CRITIC-ROLE-REQUIRED", "packets");
  const boundReview = criticReview && reviewFacts(state.physicalRoot, state, packets, criticReview, prepareDeadlineEpochMs);
  if (criticReview && (!boundReview || !reviewWorktreeClean(state.physicalRoot, boundReview.reviewedPaths, prepareDeadlineEpochMs))) return rejected("AGY-CRITIC-REVIEW-DRIFT", "criticReview");

  const body = { ...artifactBody({ state, nativeSubagents, packets, expiresAtEpochMs: nowEpochMs + ttlMs, criticReview: boundReview }), callShape };
  const artifact = { ...body, artifactSha256: sha256(JSON.stringify(body)) };
  const directory = artifactDirectory(state.commonDir);
  const path = artifactPath(state.commonDir, nativeSubagents, state.physicalRoot, callShape);
  const temporary = join(directory, `.${randomUUID()}.tmp`);
  try {
    mkdirSync(directory, { recursive: true, mode: 0o700 });
    writeFileSync(temporary, `${JSON.stringify(artifact)}\n`, { encoding: "utf8", flag: "wx", mode: 0o600 });
    const publication = publishReplacingRetirable({ temporary, path, state, nativeSubagents, callShape, nowEpochMs });
    if (publication !== "published") return rejected(
      publication === "exists" ? "AGY-NATIVE-ARTIFACT-EXISTS" : "AGY-NATIVE-ARTIFACT-WRITE",
      "artifact",
    );
  } catch {
    return rejected("AGY-NATIVE-ARTIFACT-WRITE", "artifact");
  } finally {
    try { unlinkSync(temporary); } catch { /* canonical link, if published, owns the bytes */ }
  }
  return {
    schema: AGY_NATIVE_BATCH_PREPARATION_SCHEMA,
    status: "prepared",
    code: "AGY-NATIVE-PREPARED",
    artifactSha256: artifact.artifactSha256,
    nativeSubagentsSha256: sha256(JSON.stringify(nativeSubagents)),
    callShape,
    invocation: callShape === "flat"
      ? structuredClone(nativeSubagents[0])
      : { Subagents: structuredClone(nativeSubagents) },
    modelCalls: 0,
    launcherCalls: 0,
  };
}

/** Validate the exact native call against its candidate-bound prepared artifact. */
export function verifyAntigravityNativeDispatch({ root, nativeSubagents, callShape = "batch", nowEpochMs = Date.now() } = {}) {
  const prepareDeadlineEpochMs = performance.now() + PREPARE_DEADLINE_MS;
  const state = repositoryState(root, prepareDeadlineEpochMs);
  if (state === null) return rejected("AGY-NATIVE-ROOT", "root");
  if (!validNativeSubagents(nativeSubagents)) return rejected("AGY-NATIVE-SUBAGENTS", "Subagents");
  if (!["flat", "batch"].includes(callShape)) return rejected("AGY-NATIVE-CALL-SHAPE", "callShape");
  if (callShape === "flat" && nativeSubagents.length !== 1) return rejected("AGY-NATIVE-CALL-SHAPE", "Subagents");
  const tierRefusal = nativeTierRefusal(nativeSubagents);
  if (tierRefusal !== null) return rejected(tierRefusal, "Subagents.Model");
  const path = artifactPath(state.commonDir, nativeSubagents, state.physicalRoot, callShape);
  const claimedPath = join(dirname(path), `.${randomUUID()}.consume`);
  let raw;
  try {
    // Claim-before-read makes the authorization one-shot. Only one concurrent hook can move
    // the canonical name; every replay observes a missing artifact and reaches no launcher.
    renameSync(path, claimedPath);
    const stat = lstatSync(claimedPath);
    if (!stat.isFile() || stat.isSymbolicLink() || stat.size < 2 || stat.size > MAX_ARTIFACT_BYTES || realpathSync(claimedPath) !== resolve(claimedPath)) {
      return rejected("AGY-NATIVE-ARTIFACT-FILE", "artifact");
    }
    raw = readFileSync(claimedPath, "utf8");
  } catch { return rejected("AGY-NATIVE-ARTIFACT-MISSING", "artifact"); }
  finally {
    try { unlinkSync(claimedPath); } catch { /* absent before claim or already removed */ }
  }

  let artifact;
  try { artifact = JSON.parse(raw); }
  catch { return rejected("AGY-NATIVE-ARTIFACT-MALFORMED", "artifact"); }
  if (!exactKeys(artifact, artifact?.criticReview ? [...ARTIFACT_KEYS, "criticReview"] : ARTIFACT_KEYS) || artifact.schema !== AGY_NATIVE_BATCH_ARTIFACT_SCHEMA
    || !SHA256.test(artifact.artifactSha256 ?? "") || !SHA256.test(artifact.repositoryRootSha256 ?? "")
    || !Number.isSafeInteger(artifact.expiresAtEpochMs) || !exactKeys(artifact.candidate, ["commit", "tree"])) {
    return rejected("AGY-NATIVE-ARTIFACT-SHAPE", "artifact");
  }
  const { artifactSha256, ...body } = artifact;
  if (sha256(JSON.stringify(body)) !== artifactSha256) return rejected("AGY-NATIVE-ARTIFACT-DIGEST", "artifactSha256");
  if (artifact.repositoryRootSha256 !== sha256(state.physicalRoot)) return rejected("AGY-NATIVE-ARTIFACT-ROOT", "repositoryRootSha256");
  if (artifact.callShape !== callShape) return rejected("AGY-NATIVE-CALL-SHAPE", "callShape");
  if (artifact.expiresAtEpochMs < nowEpochMs) return rejected("AGY-NATIVE-ARTIFACT-EXPIRED", "expiresAtEpochMs");
  if (artifact.candidate.commit !== state.head || artifact.candidate.tree !== state.tree) return rejected("AGY-NATIVE-ARTIFACT-STALE", "candidate");
  if (JSON.stringify(artifact.nativeSubagents) !== JSON.stringify(nativeSubagents)) return rejected("AGY-NATIVE-ARRAY-MUTATED", "Subagents");
  const artifactTierRefusal = nativeTierRefusal(artifact.nativeSubagents);
  if (artifactTierRefusal !== null) return rejected(artifactTierRefusal, "Subagents.Model");
  if (!entriesMatchPackets(nativeSubagents, artifact.packets)) return rejected("AGY-NATIVE-ARRAY-MISMATCH", "Subagents");
  if (artifact.criticReview && criticPacketIndex(artifact.packets) < 0) return rejected("AGY-CRITIC-ROLE-REQUIRED", "packets");

  if (criticPacketIndex(artifact.packets) >= 0) {
    const review = artifact.criticReview;
    if (!exactKeys(review, CRITIC_REVIEW_KEYS) || !SHA256.test(review.diffSha256 ?? "") || !SHA256.test(review.packetSha256 ?? "")) return rejected("AGY-CRITIC-REVIEW-MISSING", "criticReview");
    const current = reviewFacts(state.physicalRoot, state, artifact.packets, { authorId: review.authorId, baseCommit: review.baseCommit, reviewedPaths: review.reviewedPaths }, prepareDeadlineEpochMs);
    if (!current || JSON.stringify(current) !== JSON.stringify(review) || !reviewWorktreeClean(state.physicalRoot, review.reviewedPaths, prepareDeadlineEpochMs)) return rejected("AGY-CRITIC-REVIEW-DRIFT", "criticReview");
  }

  const batch = prepareRoleDispatchBatch({
    root: state.physicalRoot,
    packets: artifact.packets,
    deadlineEpochMs: prepareDeadlineEpochMs,
  });
  if (batch.status !== "prepared") return { ...rejected("AGY-NATIVE-BATCH-STALE", "packets"), preparation: batch };
  if (artifact.packets.some((packet) => packet.transport !== "antigravity" || packet.resultDestination?.kind !== "return"
    || packet.candidate.commit !== state.head || packet.candidate.tree !== state.tree)) {
    return rejected("AGY-NATIVE-PACKET-BOUNDARY", "packets");
  }
  const verdict = {
    schema: AGY_NATIVE_BATCH_VERDICT_SCHEMA,
    status: "prepared",
    code: "AGY-NATIVE-PREPARED",
    artifactSha256,
    packets: structuredClone(artifact.packets),
    nativeSubagents: structuredClone(nativeSubagents),
    callShape,
    workspace: state.physicalRoot,
    modelCalls: 0,
    launcherCalls: 0,
  };
  if (artifact.criticReview) {
    const context = Object.freeze(Object.create(null));
    verifiedCriticContexts.set(context, { root: state.physicalRoot, artifactSha256, candidate: structuredClone(artifact.candidate), review: structuredClone(artifact.criticReview), packet: structuredClone(artifact.packets[criticPacketIndex(artifact.packets)]), child: structuredClone(nativeSubagents[criticPacketIndex(artifact.packets)]) });
    verdict.originalCriticContext = context;
  }
  return verdict;
}

/** Controlled fixtures exercise Source mechanics only. The method result is never native/provider attestation. */
export function captureControlledAgyOriginalCriticReturn({ context, childId, verdict, reportSha256, observedModel } = {}) {
  const held = verifiedCriticContexts.get(context);
  if (!held) return { ok: false, code: "AGY-CRITIC-ORIGINAL-CONTEXT-REQUIRED" };
  if (observedCriticReturns.has(context)) return { ok: false, code: "AGY-CRITIC-ORIGINAL-REPLAY" };
  if (!CHILD_ID.test(childId ?? "") || childId === held.review.authorId
    || !["PASS", "FAIL"].includes(verdict) || !SHA256.test(reportSha256 ?? "") || observedModel !== held.child.Model) return { ok: false, code: "AGY-CRITIC-ORIGINAL-CHILD" };
  const state = repositoryState(held.root);
  const review = state && reviewFacts(held.root, state, [held.packet], { authorId: held.review.authorId, baseCommit: held.review.baseCommit, reviewedPaths: held.review.reviewedPaths }, performance.now() + PREPARE_DEADLINE_MS);
  if (!review || JSON.stringify(review) !== JSON.stringify(held.review) || state.head !== held.candidate.commit || state.tree !== held.candidate.tree
    || !reviewWorktreeClean(held.root, held.review.reviewedPaths, performance.now() + PREPARE_DEADLINE_MS)) return { ok: false, code: "AGY-CRITIC-ORIGINAL-DRIFT" };
  const value = { schema: "pipeline.agy-original-critic-return.v1", assurance: "controlled-unit-method", preparedSha256: held.artifactSha256,
    dispatchId: held.packet.dispatchId, candidate: { base: held.review.baseCommit, ...structuredClone(held.candidate) },
    diff: { sha256: held.review.diffSha256, reviewedPaths: [...held.review.reviewedPaths] },
    originalDispatch: { packetSha256: held.review.packetSha256, reviewerRole: held.packet.role },
    originalChild: { childId, model: observedModel, authorId: held.review.authorId },
    reviewerResult: { verdict, reportSha256 }, verdict, reviewPass: false,
    hostReturn: { status: "controlled-unit-method", nativeProviderAttested: false } };
  observedCriticReturns.set(context, value);
  return { ok: true, status: "method-observed", code: "AGY-CRITIC-ORIGINAL-METHOD-OBSERVED" };
}

/** The installed AGY hook observes only the pre-call. A native-labelled JSON
 * return cannot supply the missing original child terminal event. */
export function captureAgyOriginalCriticNativeReturn() {
  return { ok: false, code: "AGY-CRITIC-NATIVE-RETURN-UNAVAILABLE" };
}

/** A caller's JSON cannot become a Source handle. Recheck the original facts on every read. */
export function inspectOriginalAgyCriticReturn({ context, root } = {}) {
  const held = verifiedCriticContexts.get(context), value = observedCriticReturns.get(context);
  if (!held || !value || root !== held.root) return { ok: false, code: "AGY-CRITIC-ORIGINAL-SOURCE-REQUIRED" };
  const state = repositoryState(held.root);
  const current = state && reviewFacts(held.root, state, [held.packet], { authorId: held.review.authorId, baseCommit: held.review.baseCommit, reviewedPaths: held.review.reviewedPaths }, performance.now() + PREPARE_DEADLINE_MS);
  if (!current || JSON.stringify(current) !== JSON.stringify(held.review) || state.head !== held.candidate.commit || state.tree !== held.candidate.tree
    || !reviewWorktreeClean(held.root, held.review.reviewedPaths, performance.now() + PREPARE_DEADLINE_MS)) return { ok: false, code: "AGY-CRITIC-ORIGINAL-DRIFT" };
  const handle = Object.freeze(Object.create(null));
  inspectedCriticReturns.set(handle, { context, value });
  return { ok: true, handle, assurance: value.assurance };
}

export function readInspectedOriginalAgyCriticReturn(handle) {
  const held = inspectedCriticReturns.get(handle);
  if (!held) return { ok: false, code: "AGY-CRITIC-ORIGINAL-HANDLE-REQUIRED" };
  const source = verifiedCriticContexts.get(held.context);
  const checked = inspectOriginalAgyCriticReturn({ context: held.context, root: source?.root });
  return checked.ok ? { ok: true, value: structuredClone(held.value) } : checked;
}
