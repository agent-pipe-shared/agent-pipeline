import assert from "node:assert/strict";
import { execFileSync } from "node:child_process";
import { createHash } from "node:crypto";
import { mkdirSync, mkdtempSync, rmSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import test from "node:test";
import {
  captureAgyOriginalCriticNativeReturn,
  captureControlledAgyOriginalCriticReturn,
  inspectOriginalAgyCriticReturn,
  prepareAntigravityNativeDispatch,
  readInspectedOriginalAgyCriticReturn,
  verifyAntigravityNativeDispatch,
} from "./antigravity-native-dispatch-coordinator.mjs";
import { ROLE_DISPATCH_REQUEST_SCHEMA } from "./role-dispatch-preflight.mjs";

const digest = value => createHash("sha256").update(value).digest("hex");
const git = (root, ...args) => execFileSync("git", args, { cwd: root, encoding: "utf8", shell: false }).trim();

function fixture() {
  const parent = process.env.PIPELINE_TEST_TMPDIR ?? join(process.cwd(), "scratch");
  mkdirSync(parent, { recursive: true });
  const root = mkdtempSync(join(parent, "agy-original-critic-"));
  git(root, "init", "-q");
  git(root, "config", "user.name", "Synthetic Fixture");
  git(root, "config", "user.email", "fixture@example.invalid");
  writeFileSync(join(root, "input-a.txt"), "base\n");
  git(root, "add", "input-a.txt");
  git(root, "commit", "-q", "-m", "synthetic base");
  const baseCommit = git(root, "rev-parse", "HEAD");
  writeFileSync(join(root, "input-a.txt"), "candidate\n");
  git(root, "add", "input-a.txt");
  git(root, "commit", "-q", "-m", "synthetic candidate");
  const candidate = { commit: git(root, "rev-parse", "HEAD"), tree: git(root, "rev-parse", "HEAD^{tree}") };
  const child = { TypeName: "critic", Role: "Critic", Prompt: "Review input-a.txt independently. Model: gemini-pro; Ruleset-SHA: synthetic.\n- Tool budget: 30 tool uses", Model: "pro" };
  const packet = { schema: ROLE_DISPATCH_REQUEST_SCHEMA, dispatchId: "agy-critic-original", transport: "antigravity",
    role: "pipeline-core:critic", prompt: child.Prompt, candidate, requiredPaths: ["input-a.txt"],
    requiredPathSha256: { "input-a.txt": digest("candidate\n") }, resultDestination: { kind: "return" } };
  return { root, baseCommit, candidate, child, packet, criticReview: { authorId: "synthetic-author", baseCommit, reviewedPaths: ["input-a.txt"] } };
}

function withFixture(run) {
  const value = fixture();
  try { return run(value); }
  finally { rmSync(value.root, { recursive: true, force: true }); }
}

function verified(value) {
  const args = { root: value.root, packets: [value.packet], nativeSubagents: [value.child], callShape: "flat", criticReview: value.criticReview };
  const prepared = prepareAntigravityNativeDispatch(args);
  assert.equal(prepared.status, "prepared", JSON.stringify(prepared));
  const result = verifyAntigravityNativeDispatch({ root: value.root, nativeSubagents: [value.child], callShape: "flat" });
  assert.equal(result.status, "prepared", JSON.stringify(result));
  assert.equal(typeof result.originalCriticContext, "object");
  return result;
}

test("original AGY Critic METHOD return binds prepared packet, diff, candidate and child through an opaque Source handle", () => withFixture(value => {
  const result = verified(value);
  const context = result.originalCriticContext;
  assert.equal(inspectOriginalAgyCriticReturn({ context, root: value.root }).code, "AGY-CRITIC-ORIGINAL-SOURCE-REQUIRED");
  const captured = captureControlledAgyOriginalCriticReturn({ context, childId: "synthetic-child-1", observedModel: "pro", verdict: "PASS", reportSha256: digest("synthetic report") });
  assert.equal(captured.status, "method-observed");
  const inspected = inspectOriginalAgyCriticReturn({ context, root: value.root });
  assert.equal(inspected.ok, true, JSON.stringify(inspected));
  const read = readInspectedOriginalAgyCriticReturn(inspected.handle);
  assert.equal(read.ok, true);
  assert.equal(read.value.assurance, "controlled-unit-method");
  assert.equal(read.value.candidate.commit, value.candidate.commit);
  assert.equal(read.value.candidate.tree, value.candidate.tree);
  assert.equal(read.value.candidate.base, value.baseCommit);
  assert.equal(read.value.originalDispatch.packetSha256, digest(JSON.stringify(value.packet)));
  assert.equal(read.value.originalChild.childId, "synthetic-child-1");
  assert.equal(read.value.verdict, "PASS");
  assert.equal(read.value.reviewPass, false, "METHOD evidence cannot become independent PASS");
  assert.equal(read.value.hostReturn.nativeProviderAttested, false);
  assert.equal(readInspectedOriginalAgyCriticReturn(structuredClone(inspected.handle)).code, "AGY-CRITIC-ORIGINAL-HANDLE-REQUIRED");
  assert.equal(inspectOriginalAgyCriticReturn({ context: JSON.parse(JSON.stringify(context)), root: value.root }).code, "AGY-CRITIC-ORIGINAL-SOURCE-REQUIRED");
  assert.equal(captureControlledAgyOriginalCriticReturn({ context, childId: "synthetic-child-2", observedModel: "pro", verdict: "PASS", reportSha256: digest("other") }).code, "AGY-CRITIC-ORIGINAL-REPLAY");
}));

test("missing Critic review, foreign child model and invalid return cannot create Source admission", () => withFixture(value => {
  const absent = prepareAntigravityNativeDispatch({ root: value.root, packets: [value.packet], nativeSubagents: [value.child], callShape: "flat" });
  assert.equal(absent.code, "AGY-CRITIC-REVIEW-REQUIRED");
  const context = verified(value).originalCriticContext;
  assert.equal(captureControlledAgyOriginalCriticReturn({ context, childId: "synthetic-author", observedModel: "pro", verdict: "PASS", reportSha256: digest("report") }).code, "AGY-CRITIC-ORIGINAL-CHILD");
  assert.equal(captureControlledAgyOriginalCriticReturn({ context, childId: "child", observedModel: "flash", verdict: "PASS", reportSha256: digest("report") }).code, "AGY-CRITIC-ORIGINAL-CHILD");
  assert.equal(captureControlledAgyOriginalCriticReturn({ context, childId: "child", observedModel: "pro", verdict: "PASS", reportSha256: "forged" }).code, "AGY-CRITIC-ORIGINAL-CHILD");
  assert.equal(readInspectedOriginalAgyCriticReturn({}).code, "AGY-CRITIC-ORIGINAL-HANDLE-REQUIRED");
  assert.equal(captureAgyOriginalCriticNativeReturn({ schema: "native", verdict: "PASS" }).code, "AGY-CRITIC-NATIVE-RETURN-UNAVAILABLE");
}));

test("changed reviewed bytes after return invalidate the Source readback", () => withFixture(value => {
  const context = verified(value).originalCriticContext;
  assert.equal(captureControlledAgyOriginalCriticReturn({ context, childId: "child", observedModel: "pro", verdict: "PASS", reportSha256: digest("report") }).ok, true);
  const handle = inspectOriginalAgyCriticReturn({ context, root: value.root }).handle;
  writeFileSync(join(value.root, "input-a.txt"), "changed after return\n");
  assert.equal(readInspectedOriginalAgyCriticReturn(handle).code, "AGY-CRITIC-ORIGINAL-DRIFT");
}));
