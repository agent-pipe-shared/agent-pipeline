/** Pure request shape and digest. Keep the Spec parser outside Source readback's import graph. */
import { createHash } from "node:crypto";
import { canonical } from "./po-approval-proof.mjs";

export const DESIGN_REVIEW_EXECUTION_REQUEST_SCHEMA = "pipeline.design-review-execution-request.proposed.v1";
export const DESIGN_REVIEW_EXECUTION_ASSURANCE = Object.freeze({ class: "proposed-preparation", native: false, independentCriticPass: false });
const sha = value => createHash("sha256").update(value).digest("hex");
const hash = value => typeof value === "string" && /^[a-f0-9]{64}$/u.test(value);
const oid = value => typeof value === "string" && /^[a-f0-9]{40}$/u.test(value);
const keys = (value, expected) => value !== null && typeof value === "object" && !Array.isArray(value)
  && Object.keys(value).length === expected.length && expected.every(key => Object.hasOwn(value, key));
const pathOk = value => typeof value === "string" && value.length > 0 && value.length <= 240 && value === value.trim()
  && !/[\\\x00-\x1f:]/u.test(value) && !value.split("/").some(part => !part || part === "." || part === "..");
const refuse = code => ({ ok: false, code });
const sourceKeys = ["path", "sha256", "blobOid", "mode"];
const requestKeys = ["schema", "status", "featureId", "candidate", "sources", "specPath", "criteria", "authorExecutionSha256", "reviewerDuty", "round", "previousReceiptSha256", "assurance"];

function validateRequest(request) {
  if (!keys(request, requestKeys) || request.schema !== DESIGN_REVIEW_EXECUTION_REQUEST_SCHEMA || request.status !== "proposed"
    || !/^[a-z][a-z0-9-]{0,63}$/u.test(request.featureId ?? "")
    || !keys(request.candidate, ["commit", "tree"]) || !oid(request.candidate.commit) || !/^(?:[a-f0-9]{40}|[a-f0-9]{64})$/u.test(request.candidate.tree ?? "")
    || !Array.isArray(request.sources) || request.sources.length < 2 || request.sources.length > 66
    || !hash(request.authorExecutionSha256) || request.reviewerDuty !== "critic"
    || ![1, 2].includes(request.round) || (request.round === 1 ? request.previousReceiptSha256 !== null : !hash(request.previousReceiptSha256))
    || canonical(request.assurance) !== canonical(DESIGN_REVIEW_EXECUTION_ASSURANCE)) return refuse("DRE-REQUEST-SHAPE");
  for (let index = 0; index < request.sources.length; index++) {
    const source = request.sources[index];
    if (!keys(source, sourceKeys) || !pathOk(source.path) || !hash(source.sha256)
      || !/^(?:[a-f0-9]{40}|[a-f0-9]{64})$/u.test(source.blobOid ?? "") || !["100644", "100755"].includes(source.mode)
      || index > 0 && request.sources[index - 1].path >= source.path) return refuse("DRE-SOURCES");
  }
  const spec = request.sources.find(source => source.path === request.specPath);
  if (!spec || !keys(request.criteria, ["specSha256", "ids", "sha256"]) || request.criteria.specSha256 !== spec.sha256
    || !Array.isArray(request.criteria.ids) || !request.criteria.ids.length || request.criteria.ids.length > 4096
    || request.criteria.ids.some((id, index) => typeof id !== "string" || !/^AC-(?:[1-9][0-9]*|E[1-9][0-9]*-[1-9][0-9]*-[0-9]{2})$/u.test(id)
      || index > 0 && request.criteria.ids[index - 1] >= id)
    || request.criteria.sha256 !== sha(canonical({ specSha256: spec.sha256, ids: request.criteria.ids }))) return refuse("DRE-CRITERIA");
  return { ok: true, requestSha256: sha(canonical(request)) };
}

export function validateDesignReviewExecutionRequest(request) {
  try { return validateRequest(request); } catch { return refuse("DRE-REQUEST-SHAPE"); }
}
