// Controlled METHOD qualification: a real Node hook process runs in a production-generated view.
import test from 'node:test';
import assert from 'node:assert/strict';
import { execFileSync, spawn } from 'node:child_process';
import { createHash } from 'node:crypto';
import { appendFileSync, cpSync, existsSync, mkdirSync, mkdtempSync, readFileSync, readdirSync, rmSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { refreshAntigravityGlobalFromRunnerView } from '../../install-agy.mjs';
import { pruneAgyRunnerViews, publishAgyRunnerView } from '../agy-central-snapshot.mjs';

const NAME = 'agent-pipeline-core';
const HOOK = 'hooks/controlled-hook.mjs';
const agent = '---\nname: sample\ndescription: fixture\nmodel: sonnet\n---\n';
const digest = (path) => createHash('sha256').update(readFileSync(path)).digest('hex');
const hook = `import {writeFileSync,existsSync} from 'node:fs';
import {join} from 'node:path';
const [ready, release, done] = process.argv.slice(2);
writeFileSync(ready, JSON.stringify({cwd:process.cwd(),script:import.meta.url})+'\\n');
while (!existsSync(release)) await new Promise(resolve => setTimeout(resolve, 20));
writeFileSync(done, JSON.stringify({cwd:process.cwd(),script:import.meta.url})+'\\n');
`;
const runGit = (repo, args) => execFileSync('git', ['-C', repo, ...args], { stdio: 'ignore' });
const waitFor = async (path, child) => {
  for (let i = 0; i < 200; i++) {
    if (existsSync(path)) return JSON.parse(readFileSync(path, 'utf8'));
    if (child.exitCode !== null) throw new Error(`hook exited before ${path}: ${child.exitCode}`);
    await new Promise(resolve => setTimeout(resolve, 20));
  }
  throw new Error(`hook did not write ${path}`);
};
const settle = (child) => new Promise((resolve, reject) => {
  child.once('error', reject);
  child.once('exit', (code, signal) => resolve({ code, signal }));
});

function fixture(t) {
  const base = mkdtempSync(join(tmpdir(), 'agy-active-cwd-'));
  const home = join(base, 'home');
  const configRoot = join(home, '.gemini');
  const workspaceRoot = join(base, 'workspace');
  const repo = join(base, 'source');
  const pluginRoot = join(repo, 'plugins', 'pipeline-core');
  const managed = join(configRoot, 'config', 'plugins', NAME);
  const registry = join(configRoot, 'config', 'plugins.json');
  const imports = join(configRoot, 'config', 'import_manifest.json');
  for (const dir of [join(configRoot, 'config', 'plugins'), workspaceRoot, join(pluginRoot, 'agents'), join(pluginRoot, 'hooks'), join(pluginRoot, 'scripts'), join(pluginRoot, 'skills', 'critic-review'), join(pluginRoot, 'lib')]) mkdirSync(dir, { recursive: true });
  writeFileSync(join(pluginRoot, 'plugin.json'), `{"name":"${NAME}","version":"0.7.0","description":"Synthetic test plugin"}\n`);
  writeFileSync(join(pluginRoot, 'hooks.json'), '{"pipeline-core":{"PreInvocation":[{"type":"command","command":"node hooks/antigravity-start-hint.mjs"}]}}\n');
  for (const path of ['hooks/antigravity-pretool-guard.mjs', 'hooks/antigravity-start-hint.mjs', 'skills/critic-review/SKILL.md', 'scripts/pipeline-start-preflight.mjs', 'lib/example.test.mjs']) writeFileSync(join(pluginRoot, path), '// synthetic fixture\n');
  writeFileSync(join(pluginRoot, 'agents', 'a.md'), agent);
  writeFileSync(join(pluginRoot, HOOK), hook);
  writeFileSync(imports, '{"imports":null}\n');
  writeFileSync(registry, '{"entries":[]}\n');
  runGit(repo, ['init', '--initial-branch', 'main']);
  runGit(repo, ['remote', 'add', 'origin', 'https://example.invalid/synthetic.git']);
  runGit(repo, ['add', '.']);
  runGit(repo, ['-c', 'user.name=Fixture', '-c', 'user.email=fixture@example.test', 'commit', '-m', 'fixture']);
  t.after(() => rmSync(base, { recursive: true, force: true }));
  const cliCalls = [];
  const runCli = (argv) => {
    cliCalls.push([...argv]);
    if (argv[0] === '--version') return { status: 'ok', version: '1.4.2' };
    if (argv[1] === 'install') {
      cpSync(argv[2], managed, { recursive: true });
      writeFileSync(imports, `{"imports":[{"name":"${NAME}","source":${JSON.stringify(argv[2])}}]}\n`);
    }
    if (argv[1] === 'uninstall') {
      rmSync(managed, { recursive: true, force: true });
      writeFileSync(imports, '{"imports":null}\n');
    }
    return { status: 'ok' };
  };
  const options = { configRoot, workspaceRoot, approvedSourceRoot: pluginRoot, attestationSourceRoot: pluginRoot, scope: 'global', globalChangeApproved: true };
  const deps = { runCli, snapshotDeps: { homedirFn: () => home }, writeInstalledReceipt: () => ({ status: 'written' }) };
  const refresh = () => refreshAntigravityGlobalFromRunnerView(options, deps);
  return { base, home, repo, pluginRoot, configRoot, workspaceRoot, managed, registry, imports, cliCalls, refresh, deps };
}

test('G9b: old hook cwd and script survive a production refresh from a revised synthetic Source', async (t) => {
  const w = fixture(t);
  const first = w.refresh();
  assert.equal(first.status, 'refreshed', JSON.stringify(first));
  const oldRoot = w.cliCalls.find(args => args[1] === 'install')[2];
  const oldScript = join(oldRoot, HOOK);
  const sourceFile = join(w.pluginRoot, 'agents', 'a.md');
  const firstSourceSha256 = digest(sourceFile);
  assert.equal(existsSync(oldScript), true);
  const ready = join(w.base, 'ready.json'), release = join(w.base, 'release'), done = join(w.base, 'done.json');
  const child = spawn(process.execPath, [oldScript, ready, release, done], { cwd: oldRoot, stdio: 'ignore', env: { ...process.env } });
  let settled = false;
  t.after(() => { if (!settled) child.kill('SIGKILL'); });
  try {
    const started = await waitFor(ready, child);
    assert.equal(started.cwd, oldRoot);
    assert.equal(started.script, new URL(`file://${oldScript}`).href);
    appendFileSync(join(w.pluginRoot, 'agents', 'a.md'), 'revised source\n');
    runGit(w.repo, ['add', '.']);
    runGit(w.repo, ['-c', 'user.name=Fixture', '-c', 'user.email=fixture@example.test', 'commit', '-m', 'revision']);
    const sourceRevision = execFileSync('git', ['-C', w.repo, 'rev-parse', 'HEAD'], { encoding: 'utf8' }).trim();
    const secondSourceSha256 = digest(sourceFile);
    assert.notEqual(secondSourceSha256, firstSourceSha256);
    const second = w.refresh();
    assert.equal(second.status, 'refreshed', JSON.stringify(second));
    assert.notEqual(second.runnerView.viewSha256, first.runnerView.viewSha256);
    assert.equal(second.runnerView.pruneSkipped, 'AGV-PRUNE-LIVENESS-UNOBSERVABLE');
    assert.deepEqual(second.runnerView.removed, []);
    assert.equal(existsSync(oldRoot), true);
    assert.equal(existsSync(oldScript), true);
    writeFileSync(release, 'go\n');
    const completion = await settle(child);
    settled = true;
    assert.deepEqual(completion, { code: 0, signal: null });
    const ended = JSON.parse(readFileSync(done, 'utf8'));
    assert.deepEqual(ended, started);
    const later = spawn(process.execPath, [oldScript, join(w.base, 'later-ready.json'), release, join(w.base, 'later-done.json')], { cwd: oldRoot, stdio: 'ignore', env: { ...process.env } });
    const laterExit = await settle(later);
    assert.deepEqual(laterExit, { code: 0, signal: null });
    assert.equal(JSON.parse(readFileSync(join(w.base, 'later-done.json'), 'utf8')).cwd, oldRoot);
    assert.equal(readdirSync(join(w.home, '.agent-pipeline', 'agy-runner-view')).filter(name => name.startsWith('view-')).length, 2);
    console.log(JSON.stringify({ case: 'active-cwd-refresh', sourceRevision, firstSourceSha256, secondSourceSha256, firstViewSha256: first.runnerView.viewSha256, secondViewSha256: second.runnerView.viewSha256, hookExit: completion.code, laterExit: laterExit.code, cliCalls: w.cliCalls.length, retainedViewCount: 2 }));
  } finally {
    if (!settled) { child.kill('SIGKILL'); await settle(child); }
  }
});

test('G9b: direct custody prune without complete liveness observation can remove a live cwd; foreign tree is retained', async (t) => {
  const w = fixture(t);
  const view = publishAgyRunnerView({ sourcePluginRoot: w.pluginRoot, deps: w.deps.snapshotDeps });
  const foreign = join(w.home, '.agent-pipeline', 'agy-runner-view', `view-${'f'.repeat(64)}`);
  mkdirSync(foreign);
  writeFileSync(join(foreign, 'foreign.txt'), 'retain\n');
  const ready = join(w.base, 'ready.json'), release = join(w.base, 'release'), done = join(w.base, 'done.json');
  const child = spawn(process.execPath, [join(view.root, HOOK), ready, release, done], { cwd: view.root, stdio: 'ignore', env: { ...process.env } });
  let settled = false;
  t.after(() => { if (!settled) child.kill('SIGKILL'); });
  try {
    assert.equal((await waitFor(ready, child)).cwd, view.root);
    const pruned = pruneAgyRunnerViews({ keepPaths: [], keepDigests: [], deps: w.deps.snapshotDeps });
    assert.deepEqual(pruned.removed, [`view-${view.viewSha256}`]);
    assert.deepEqual(pruned.retained, [`view-${'f'.repeat(64)}`]);
    assert.equal(existsSync(view.root), false);
    assert.equal(existsSync(foreign), true);
    writeFileSync(release, 'go\n');
    assert.deepEqual(await settle(child), { code: 0, signal: null });
    settled = true;
    assert.equal(existsSync(done), true, 'a Unix process can finish from an unlinked cwd, but its next script call cannot');
    console.log(JSON.stringify({ case: 'direct-prune-limit', viewSha256: view.viewSha256, removedCount: pruned.removed.length, retainedForeignCount: pruned.retained.length, hookExit: child.exitCode }));
  } finally {
    if (!settled) { child.kill('SIGKILL'); await settle(child); }
  }
});
