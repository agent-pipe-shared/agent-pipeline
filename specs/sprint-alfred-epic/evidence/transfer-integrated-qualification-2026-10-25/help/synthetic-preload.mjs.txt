import os from 'node:os';
import { syncBuiltinESMExports } from 'node:module';
import { lstatSync, realpathSync, mkdtempSync, chmodSync, rmSync } from 'node:fs';
import { join } from 'node:path';
process.env.NODE_OPTIONS = `--import=${import.meta.url}`;
const parent = '/run/user/1000/ap-g17';
function validate(path) {
  const st=lstatSync(path);
  if(!st.isDirectory()||st.isSymbolicLink()||realpathSync(path)!==path||st.uid!==process.getuid()||(st.mode&0o077))throw new Error('synthetic directory custody unavailable');
  return st;
}
validate(parent);
const owner=!process.env.R2_HELP_HOME;
const home=process.env.R2_HELP_HOME??mkdtempSync(join(parent,'r2-help-'));
const temp=process.env.R2_HELP_TMP??mkdtempSync('/tmp/r2-help-');
chmodSync(home,0o700);chmodSync(temp,0o700);
const held=validate(home),tempHeld=validate(temp);
Object.assign(process.env,{R2_HELP_HOME:home,R2_HELP_TMP:temp,HOME:home,USERPROFILE:home,XDG_CONFIG_HOME:join(home,'config'),XDG_DATA_HOME:join(home,'data'),GIT_CONFIG_NOSYSTEM:'1',GIT_CONFIG_GLOBAL:'/dev/null',GIT_AUTHOR_NAME:'Test',GIT_AUTHOR_EMAIL:'test@example.invalid',GIT_COMMITTER_NAME:'Test',GIT_COMMITTER_EMAIL:'test@example.invalid'});
os.homedir=()=>home;os.tmpdir=()=>temp;os.userInfo=()=>({username:'synthetic-test',uid:process.getuid(),gid:process.getgid(),homedir:home,shell:'/bin/false'});
syncBuiltinESMExports();
if(owner)process.on('exit',()=>{for(const[path,before]of[[temp,tempHeld],[home,held]]){const now=lstatSync(path);if(now.dev===before.dev&&now.ino===before.ino)rmSync(path,{recursive:true});}});
