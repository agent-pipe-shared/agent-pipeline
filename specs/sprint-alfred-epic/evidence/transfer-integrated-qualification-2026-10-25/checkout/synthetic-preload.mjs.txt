import os from 'node:os';
import { syncBuiltinESMExports } from 'node:module';
import { lstatSync, realpathSync, mkdtempSync, chmodSync, rmSync } from 'node:fs';
import { join } from 'node:path';
const parent = '/run/user/1000/ap-g17';
process.env.NODE_OPTIONS = `--import=${import.meta.url}`;
function privateDirectory(path) {
  const st = lstatSync(path);
  if (!st.isDirectory() || st.isSymbolicLink() || realpathSync(path) !== path || st.uid !== process.getuid() || (st.mode & 0o077)) throw new Error('test home custody unavailable');
  return st;
}
privateDirectory(parent);
const owner = !process.env.R2_TEST_HOME;
const home = process.env.R2_TEST_HOME ?? mkdtempSync(join(parent, 'r2-checkout-'));
chmodSync(home, 0o700);
const held = privateDirectory(home);
const temp = process.env.R2_TEST_TMP ?? mkdtempSync('/tmp/r2-checkout-');
chmodSync(temp, 0o700);
const tmpHeld = privateDirectory(temp);
Object.assign(process.env, { R2_TEST_HOME: home, R2_TEST_TMP: temp, HOME: home, USERPROFILE: home, XDG_CONFIG_HOME: join(home,'config'), XDG_DATA_HOME: join(home,'data'), GIT_CONFIG_NOSYSTEM: '1', GIT_CONFIG_GLOBAL: '/dev/null', GIT_AUTHOR_NAME: 'Test', GIT_AUTHOR_EMAIL: 'test@example.invalid', GIT_COMMITTER_NAME: 'Test', GIT_COMMITTER_EMAIL: 'test@example.invalid' });
os.homedir = () => home;
os.tmpdir = () => temp;
os.userInfo = () => ({ username: 'synthetic-test', uid: process.getuid(), gid: process.getgid(), homedir: home, shell: '/bin/false' });
syncBuiltinESMExports();
if (owner) process.on('exit', () => {
  for (const [path, before] of [[temp,tmpHeld],[home,held]]) {
    const now = lstatSync(path);
    if (before.dev === now.dev && before.ino === now.ino) rmSync(path, { recursive: true });
  }
});
