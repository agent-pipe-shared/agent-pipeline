// Helper (untracked): run ONE test file/command in the k81 clone with a private TMPDIR, save output, print summary lines.
// usage: node scratch/k81-run.mjs <label> <file-relative-to-clone> [--script]
// With --script the file is run as a plain node script instead of `node --test`-style (node <file>).
import { spawnSync } from "node:child_process";
import { mkdirSync, writeFileSync, chmodSync } from "node:fs";
import { join, resolve } from "node:path";
import { homedir } from "node:os";
const [label, file, mode] = process.argv.slice(2);
const tmp = join(homedir(), "pipeline-test-tmp", "k81-" + label, "0");
mkdirSync(tmp, { recursive: true, mode: 0o700 }); chmodSync(tmp, 0o700);
const env = { ...process.env, TMPDIR: tmp, TMP: tmp, TEMP: tmp };
delete env.NODE_TEST_CONTEXT;
const args = [file];
const t0 = Date.now();
const r = spawnSync(process.execPath, args, { cwd: resolve("scratch/clones/k81"), encoding: "utf8", env, timeout: 3600000, maxBuffer: 256 * 1024 * 1024 });
const out = (r.stdout ?? "") + (r.stderr ?? "");
mkdirSync("scratch/agent-out", { recursive: true });
writeFileSync(`scratch/agent-out/k81-${label}.log`, out);
console.log(`exit ${r.status} signal ${r.signal} secs ${Math.round((Date.now() - t0) / 1000)} bytes ${out.length}`);
const lines = out.split("\n");
for (const l of lines) {
  if (/^ℹ (tests|suites|pass|fail|cancelled|skipped|todo)/.test(l) || /^\s*(✖|not ok)/.test(l) || /# (TODO|SKIP)/.test(l) || /^✖|failing tests/.test(l)) console.log(l.slice(0, 260));
}
if (mode === "--script" || !lines.some((l) => /^ℹ tests/.test(l))) console.log(lines.slice(-12).join("\n").slice(0, 1500));
