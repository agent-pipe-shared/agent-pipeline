// node scratch/run-in.mjs <dir> <label> <test-file>... -- runs the named test files (paths relative to <dir>) one after the other inside <dir> (main checkout: ".") with a PRIVATE TMPDIR outside every git repository (via run-quiet.mjs); prints counts and failing case names only.
import { spawnSync } from "node:child_process";
import { resolve } from "node:path";
const [dir, label, ...files] = process.argv.slice(2);
if (!dir || !label || files.length === 0) { console.error("usage: run-in.mjs <dir> <label> <test-file>..."); process.exit(2); }
const r = spawnSync(process.execPath, [resolve("scratch/run-quiet.mjs"), label, ...files], { cwd: resolve(dir), stdio: "inherit" });
process.exit(r.status ?? 1);
