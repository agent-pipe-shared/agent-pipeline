import {readFileSync,writeFileSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {spawnSync} from 'node:child_process';
import {resolve} from 'node:path';
const root=resolve('scratch/clones/generator-integrated-13-qualification');
const out='scratch/agent-out/generator-integrated-13-qualification';
const sha=p=>createHash('sha256').update(readFileSync(p)).digest('hex');
const git=args=>{const x=spawnSync('git',['-C',root,...args],{encoding:'utf8'});if(x.status!==0)throw Error('candidate git check failed');return x.stdout.trim();};
const candidate={commit:git(['rev-parse','HEAD']),tree:git(['show','-s','--format=%T','HEAD'])};
const paths=['plugins/pipeline-core/lib/pins/design-generator-repeats-large-source-material.test.mjs','plugins/pipeline-core/scripts/project-onboarding-v3.mjs','plugins/pipeline-core/lib/project-onboarding-v3.mjs','plugins/pipeline-core/lib/onboarding-continuity.mjs','plugins/pipeline-core/lib/onboarding-driver-output.mjs','plugins/pipeline-core/lib/test-tmpdir.mjs','plugins/pipeline-core/scripts/onboarding-init.mjs','plugins/pipeline-core/lib/governance-scope.mjs','plugins/pipeline-core/lib/machine-plane.mjs'];
const sources=Object.fromEntries(paths.map(p=>[p,sha(resolve(root,p))]));
const captures=Object.fromEntries(['original13.log','measurement.log'].map(p=>[p,sha(`${out}/${p}`)]));
const status=git(['status','--short']);
const suite=readFileSync(`${out}/original13.log`,'utf8'), measureLog=readFileSync(`${out}/measurement.log`,'utf8');
const measured=JSON.parse(measureLog.split('--- stdout ---\n')[1].split('\n--- stderr ---')[0]);
const testCounts=Object.fromEntries(['tests','pass','fail','cancelled','skipped','todo'].map(k=>[k,Number(suite.match(new RegExp('ℹ '+k+' (\\d+)'))?.[1])]));
const exit=log=>Number(log.match(/^exitCode: (\d+)$/m)?.[1]);
const originalLimit=Number(readFileSync(resolve(root,paths[0]),'utf8').match(/COLLECT_INPUT_REFERENCE_BOUND = ([\d_]+)/)[1].replaceAll('_',''));
const actualDriver=measured.drivers.codex.bytes;
const facts={schema:'pipeline.generator-qualification-source-binding.v1',candidate,cloneTrackedStatus:status,sources,captures,originalPinChanged:git(['diff','--',paths[0]])!=='',fixtureCorrection:false,testCounts,suiteExit:exit(suite),measurementExit:exit(measureLog),remainingFailure:{assertion:'bytes(stdout) <= COLLECT_INPUT_REFERENCE_BOUND',originalLimit,measuredBytes:actualDriver,headroom:originalLimit-actualDriver},qualification:exit(suite)===0?'green':'red'};
writeFileSync(`${out}/source-binding.json`,JSON.stringify(facts,null,2)+'\n');
const sourceRef={path:paths[0],sha256:sources[paths[0]],locator:'assertCompact(summary.out, 8192); COLLECT_INPUT_REFERENCE_BOUND = 15_000'};
const command='node scratch/agent-out/generator-integrated-13-qualification/run.mjs measure';
const measurement=(value)=>({value,unit:'utf8-bytes',candidate,limitSourceSha256:sourceRef.sha256,command,artifactRef:{path:`${out}/measurement.log`,sha256:captures['measurement.log']},exitCode:0});
const resources=[{id:'fixture-summary',limit:{value:8192,unit:'utf8-bytes',sourceRef},measurement:measurement(measured.measurements[0].summaryBytes)},{id:'fixture-driver-primary-reference',limit:{value:originalLimit,unit:'utf8-bytes',sourceRef},measurement:measurement(actualDriver)},{id:'native-host-output-budget',limit:null,measurement:null}];
writeFileSync(`${out}/resource-observations.json`,JSON.stringify({schema:'pipeline.project-resource-observations.v1',candidate,resources},null,2)+'\n');
console.log(JSON.stringify(facts,null,2));
